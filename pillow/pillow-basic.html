Pillow 画像の基本操作

Pillowライブラリをインストールした後、必要なモジュールをインポートする必要があります：

# 导入模块
from PIL import Image, ImageFilter, ImageEnhance, ImageDraw, ImageFont
import os

テスト画像（ローカルにダウンロードしてテスト）：

例

以下に、次の操作を示す完全な例を示します：

  1. 画像を開いて基本情報を表示
    • 形式、サイズ、モードなど
  2. 画像サイズの調整
    • 画像を指定されたサイズに調整
  3. 画像のトリミング
    • 元画像から特定の領域を抽出
  4. 画像の回転
    • 45度回転させ、画像全体を保持
  5. 画像の反転
    • 水平方向と垂直方向に反転
  6. カラーモードの変換
    • カラー画像をグレースケール画像に変換
  7. フィルターの適用
    • ぼかし、エッジ強調、輪郭などの効果
  8. 画像属性の調整
    • 明るさ、コントラスト、シャープネスの調整
  9. テキストの追加
    • 画像上に文字を描画
  10. 画像の合成
    • 簡単な画像の結合
  11. 異なる形式と品質で保存
    • PNG、BMP、さまざまな品質のJPEG

例

"""
Python Pillow 画像基本操作チュートリアル
使用するテスト画像: tiger.jpeg
"""

from PIL import Image, ImageFilter, ImageEnhance, ImageDraw, ImageFont
import os

def main():
    # 作業ディレクトリに出力フォルダーがあることを確認
    output_dir = "output"
    if not os.path.exists(output_dir):
        os.makedirs(output_dir)
   
    # 1. 画像を開く
    print("1. 画像を開く")
    try:
        img = Image.open("tiger.jpeg")
        print(f"画像を正常に開きました - 形式: {img.format}, サイズ: {img.size}, モード: {img.mode}")
    except Exception as e:
        print(f"画像を開けませんでした: {e}")
        return
   
    # 2. 画像の基本情報を表示
    print("\n2. 画像基本情報")
    print(f"画像形式: {img.format}")
    print(f"画像サイズ (幅x高さ): {img.size}")
    print(f"画像モード: {img.mode}")
   
    # 3. 画像サイズの調整
    print("\n3. 画像サイズの調整")
    resized_img = img.resize((400, 300))
    resized_img.save(f"{output_dir}/resized_tiger.jpg")
    print("サイズ調整後の画像を保存しました")
   
    # 4. 画像のトリミング
    print("\n4. 画像のトリミング")
    # トリミング領域 (左, 上, 右, 下)
    width, height = img.size
    crop_area = (width//4, height//4, 3*width//4, 3*height//4)  # 中央領域をトリミング
    cropped_img = img.crop(crop_area)
    cropped_img.save(f"{output_dir}/cropped_tiger.jpg")
    print(f"画像領域 {crop_area} をトリミングして保存しました")
   
    # 5. 画像の回転
    print("\n5. 画像の回転")
    rotated_img = img.rotate(45, expand=True)  # 45度回転、expand=True で画像全体を保持
    rotated_img.save(f"{output_dir}/rotated_tiger.jpg")
    print("回転後の画像を保存しました")
   
    # 6. 画像の反転
    print("\n6. 画像の反転")
    # 水平方向に反転
    h_flipped = img.transpose(Image.FLIP_LEFT_RIGHT)
    h_flipped.save(f"{output_dir}/horizontal_flip_tiger.jpg")
    # 垂直方向に反転
    v_flipped = img.transpose(Image.FLIP_TOP_BOTTOM)
    v_flipped.save(f"{output_dir}/vertical_flip_tiger.jpg")
    print("水平方向と垂直方向に反転した画像を保存しました")
   
    # 7. カラーモードの変換
    print("\n7. カラーモードの変換")
    grayscale_img = img.convert('L')  # グレースケール画像に変換
    grayscale_img.save(f"{output_dir}/grayscale_tiger.jpg")
    print("グレースケール画像を保存しました")
   
    # 8. フィルターの適用
    print("\n8. フィルターの適用")
    # ぼかしフィルター
    blurred_img = img.filter(ImageFilter.BLUR)
    blurred_img.save(f"{output_dir}/blurred_tiger.jpg")
    # エッジ強調
    edge_img = img.filter(ImageFilter.EDGE_ENHANCE)
    edge_img.save(f"{output_dir}/edge_enhanced_tiger.jpg")
    # 輪郭フィルター
    contour_img = img.filter(ImageFilter.CONTOUR)
    contour_img.save(f"{output_dir}/contour_tiger.jpg")
    print("異なるフィルターを適用した画像を保存しました")
   
    # 9. 明るさ、コントラスト、シャープネスの調整
    print("\n9. 画像属性の調整")
    # 明るさを増加
    brightness = ImageEnhance.Brightness(img)
    bright_img = brightness.enhance(1.5)  # 明るさを50%増加
    bright_img.save(f"{output_dir}/brighter_tiger.jpg")
   
    # コントラストを増加
    contrast = ImageEnhance.Contrast(img)
    contrast_img = contrast.enhance(1.5)  # コントラストを50%増加
    contrast_img.save(f"{output_dir}/contrast_tiger.jpg")
   
    # シャープネスを増加
    sharpness = ImageEnhance.Sharpness(img)
    sharp_img = sharpness.enhance(2.0)  # シャープネスを100%増加
    sharp_img.save(f"{output_dir}/sharper_tiger.jpg")
    print("明るさ、コントラスト、シャープネスを調整した画像を保存しました")
   
    # 10. テキストの追加
    print("\n10. テキストの追加")
    # 描画用のコピーを作成
    text_img = img.copy()
    draw = ImageDraw.Draw(text_img)
    # システムフォントを試し、失敗した場合はデフォルトを使用
    try:
        # OSによっては、異なるフォントパスが必要になる場合があります
        # Windowsのフォントパス例
        font = ImageFont.truetype("arial.ttf", 36)
    except IOError:
        font = ImageFont.load_default()
   
    # 画像上にテキストを追加
    draw.text((10, 10), "Tiger", fill=(255, 255, 255), font=font)
    text_img.save(f"{output_dir}/text_tiger.jpg")
    print("テキストを追加した画像を保存しました")
   
    # 11. 画像の合成（簡単な結合）
    print("\n11. 画像の合成")
    # 幅が元画像の2倍で、高さが同じ新規画像を作成
    merged_img = Image.new('RGB', (width*2, height))
    # 元画像を左側に配置
    merged_img.paste(img, (0, 0))
    # グレースケール画像を右側に配置
    merged_img.paste(grayscale_img.convert('RGB'), (width, 0))
    merged_img.save(f"{output_dir}/merged_tiger.jpg")
    print("合成後の画像を保存しました")
   
    # 12. 異なる形式で保存
    print("\n12. 異なる形式で保存")
    img.save(f"{output_dir}/tiger.png")  # PNG形式
    img.save(f"{output_dir}/tiger.bmp")  # BMP形式
   
    # JPEGで保存、品質を設定
    img.save(f"{output_dir}/tiger_high_quality.jpg", quality=95)  # 高品質
    img.save(f"{output_dir}/tiger_low_quality.jpg", quality=10)   # 低品質
    print("異なる形式と品質の画像を保存しました")
   
    print("\nすべての操作が完了しました！処理後の画像は 'output' フォルダーに保存されています。")

if __name__ == "__main__":
    main()

コードは構造化された方式を採用しており、各操作に明確なコメント説明があります。処理後のすべての画像は、次の場所に保存されます。outputディレクトリ配下に保存され、結果の確認と比較が容易です。

使用方法：

  • コードがあるディレクトリに tiger.jpeg 画像があることを確認してください
  • コードを実行すると、output フォルダーにさまざまな処理後の画像が生成されます
  • 出力された情報に基づいて、さまざまな処理効果を確認できます

上記のコードが正常に実行された場合、端末の出力結果は次のようになります：

操作説明

1. 画像を開いて基本情報を取得する

例

# 画像ファイルを開く
img = Image.open("tiger.jpeg")

# 画像情報を取得して表示
print(f"画像形式: {img.format}")  # 例: JPEG, PNGなど
print(f"画像サイズ (幅x高さ): {img.size}")  # 例: (800, 600)
print(f"画像モード: {img.mode}")  # 例: RGB, RGBA, L(グレースケール)など

詳細説明：

  • Image.open()関数は画像ファイルを開くために使用され、引数は画像ファイルのパスです。
  • img.format画像の形式（JPEG, PNG, GIFなど）を表示します
  • img.size幅と高さを含むタプルを返します。例: (800, 600)
  • img.mode画像のカラーモードを表示します：
    • RGB: カラー画像
    • RGBA: 透明チャンネルを持つカラー画像
    • L: グレースケール画像
    • 1: 2値画像（白黒）

2. 画像サイズの調整

例

# 画像サイズを400x300ピクセルに調整
resized_img = img.resize((400, 300))

# サイズ調整後の画像を保存
resized_img.save("output/resized_tiger.jpg")

詳細説明：

  • resize()メソッドは画像サイズを変更するために使用され、引数は新しい幅と高さを含むタプルです。
  • このメソッドは元の画像を変更せず、新しい画像オブジェクトを返します。
  • デフォルトでは最近傍補間法を使用します。より高品質な調整には、次のパラメータを追加できます：resized_img = img.resize((400, 300), Image.LANCZOS)
  • save()メソッドは処理後の画像を指定されたパスに保存します。

3. 画像のトリミング

例

# 元画像のサイズを取得
width, height = img.size

# トリミング領域を定義 (左, 上, 右, 下)
crop_area = (width//4, height//4, 3*width//4, 3*height//4)  # 中央領域をトリミング

# トリミングを実行
cropped_img = img.crop(crop_area)

# トリミング後の画像を保存
cropped_img.save("output/cropped_tiger.jpg")

詳細説明：

  • crop()メソッドは画像から矩形領域を抽出するために使用されます。
  • トリミング領域はタプル(left, upper, right, lower)定義:
    • left: 左辺界のx座標
    • upper: 上辺界のy座標
    • right: 右辺界のx座標
    • lower: 下辺界のy座標
  • この例では、画像の中心部分（1/4から3/4の領域）を切り抜きました
  • width//4の中の//は整数除算で、結果が整数になることを保証します

4. 画像の回転

例

# 画像を45度回転し、キャンバスを拡張して全体を表示します
rotated_img = img.rotate(45, expand=True)

# 回転した画像を保存します
rotated_img.save("output/rotated_tiger.jpg")

詳細説明：

  • rotate()メソッドは画像を回転させるために使用され、最初のパラメータは回転角度（反時計回り、度数単位）です
  • expand=Trueパラメータは出力画像のサイズを拡張し、回転後の画像全体を収めます
  • 設定しない場合expand=True、回転後の画像は元のサイズに合うように切り取られます
  • デフォルトでは、回転中心は画像の中心点です

5. 画像の反転

例

# 水平反転（左右ミラー）
h_flipped = img.transpose(Image.FLIP_LEFT_RIGHT)
h_flipped.save("output/horizontal_flip_tiger.jpg")

# 垂直反転（上下ミラー）
v_flipped = img.transpose(Image.FLIP_TOP_BOTTOM)
v_flipped.save("output/vertical_flip_tiger.jpg")

詳細説明：

  • transpose()メソッドは画像の反転および転置操作に使用されます
  • Image.FLIP_LEFT_RIGHT定数は水平反転（左右ミラー）を表します
  • Image.FLIP_TOP_BOTTOM定数は垂直反転（上下ミラー）を表します
  • 反転以外にも、次のような他の定数を使用できますImage.ROTATE_90, Image.ROTATE_180, Image.ROTATE_270回転を行うことができます

6. カラーモードの変換

例

# 画像をグレースケールに変換します
grayscale_img = img.convert('L')

# グレースケール画像を保存します
grayscale_img.save("output/grayscale_tiger.jpg")

詳細説明：

  • convert()メソッドは画像のモード（色空間）を変更するために使用されます
  • パラメータ'L'はグレースケール画像（Luminance）への変換を表します
  • その他の一般的なモードは次のとおりです:
    • 'RGB': 3チャンネルカラー
    • 'RGBA': 4チャンネルカラー（透明度付き）
    • 'CMYK': 印刷4色モード
    • '1': 2値画像（白黒のみ）
  • モードを変換すると、ファイルサイズを削減したり、特定の処理の準備をしたりできます

7. 画像フィルターの適用

例

# ぼかしフィルターを適用します
blurred_img = img.filter(ImageFilter.BLUR)
blurred_img.save("output/blurred_tiger.jpg")

# エッジ強調フィルターを適用します
edge_img = img.filter(ImageFilter.EDGE_ENHANCE)
edge_img.save("output/edge_enhanced_tiger.jpg")

# 輪郭フィルターを適用します
contour_img = img.filter(ImageFilter.CONTOUR)
contour_img.save("output/contour_tiger.jpg")

詳細説明：

  • filter()メソッドは画像に様々な定義済みフィルター効果を適用するために使用されます
  • ImageFilter.BLURは画像をぼかし、ソフトフォーカス効果に似ています
  • ImageFilter.EDGE_ENHANCEは画像のエッジを強調し、輪郭をより明確にします
  • ImageFilter.CONTOURは画像の輪郭線を探し、スケッチに似た効果を生み出します
  • その他の一般的なフィルターには次のものがあります:
    • ImageFilter.SHARPEN: 画像をシャープにする
    • ImageFilter.EMBOSS: エンボス効果
    • ImageFilter.FIND_EDGES: エッジ検出
    • ImageFilter.SMOOTH: 画像を滑らかにする
  • ガウスぼかしなどのカスタムフィルターを作成することもできます:img.filter(ImageFilter.GaussianBlur(radius=2))

8. 画像属性の調整（明るさ、コントラスト、シャープネス）

例

# 明るさを上げる（1.5は明るさが50%増加）
brightness = ImageEnhance.Brightness(img)
bright_img = brightness.enhance(1.5)
bright_img.save("output/brighter_tiger.jpg")

# コントラストを上げる（1.5はコントラストが50%増加）
contrast = ImageEnhance.Contrast(img)
contrast_img = contrast.enhance(1.5)
contrast_img.save("output/contrast_tiger.jpg")

# シャープネスを上げる（2.0はシャープネスが100%増加）
sharpness = ImageEnhance.Sharpness(img)
sharp_img = sharpness.enhance(2.0)
sharp_img.save("output/sharper_tiger.jpg")

詳細説明：

  • ImageEnhanceモジュールは画像のさまざまな属性を調整するためのいくつかのエンハンサークラスを提供します
  • 各エンハンサーはまず画像オブジェクトで初期化する必要があります
  • enhance()メソッドは1つのパラメータを受け取り、強調の程度を表します:
    • 値が1.0の場合は元の画像が変わらないことを意味します
    • 値が1.0より大きい場合は強調効果（例: 1.5は50%増加）を意味します
    • 値が1.0より小さい場合は弱化効果（例: 0.5は50%減少）を意味します
  • 上記で紹介した3つのエンハンサー以外にも、ImageEnhance.Colorは色の彩度を調整するために使用されます

9. 画像にテキストを追加

例

# 描画用に画像のコピーを作成します
text_img = img.copy()

# 描画オブジェクトを作成します
draw = ImageDraw.Draw(text_img)

# フォントの読み込みを試みます（利用できない場合はデフォルトフォントを使用）
try:
    # システムフォントを36ピクセルで読み込みます
    font = ImageFont.truetype("arial.ttf", 36)
except IOError:
    # 指定したフォントが見つからない場合は、デフォルトフォントを使用します
    font = ImageFont.load_default()

# 画像にテキストを追加します
draw.text((10, 10), "Tiger", fill=(255, 255, 255), font=font)

# テキストを追加した画像を保存します
text_img.save("output/text_tiger.jpg")

詳細説明：

  • copy()メソッドは元の画像のコピーを作成し、元の画像が変更されるのを防ぎます
  • ImageDraw.Draw()画像上に描画するための描画オブジェクトを作成します
  • ImageFont.truetype()TrueTypeフォントの読み込みに使用されます:
    • 最初のパラメータはフォントファイルのパス（例: "arial.ttf"）です
    • 2番目のパラメータはフォントサイズ（ピクセル単位）です
  • draw.text()は画像上にテキストを描画するために使用されます:
    • 最初のパラメータ(10, 10)はテキストの位置座標（左上隅）です
    • 2番目のパラメータは描画するテキストの内容です
    • fillパラメータは文字の色を指定し、ここでは白です(255, 255, 255)
    • fontパラメータは使用するフォントを指定します
  • を使用してtry-exceptブロックで可能性のあるフォント読み込み失敗を処理します

10. 画像の合成（簡単な結合）

例

# 元の画像のサイズを取得します
width, height = img.size

# 幅が元の画像の2倍の新しい空白画像を作成します
merged_img = Image.new('RGB', (width*2, height))

# 元の画像を左側に貼り付けます
merged_img.paste(img, (0, 0))

# グレースケール画像をRGBに変換して右側に貼り付けます
merged_img.paste(grayscale_img.convert('RGB'), (width, 0))

# 結合した画像を保存します
merged_img.save("output/merged_tiger.jpg")

詳細説明：

  • Image.new()新しい空白画像を作成します:
    • 最初のパラメータはカラーモードで、ここでは'RGB'
    • 2番目のパラメータは画像サイズで、ここでは元の画像の幅の2倍と元の画像の高さです
  • paste()メソッドはある画像を別の画像に貼り付けます:
    • 最初のパラメータは貼り付ける画像です
    • 2番目のパラメータは貼り付け位置の座標（左上隅）です
  • convert('RGB')グレースケール画像をRGBモードに変換します。同じモードの画像のみ貼り合わせることができるためです
  • この方法は、コラージュ作成、並べて比較、フォトアルバムの作成に使用できます

11. 異なる形式と品質で保存

例

# PNG形式で保存します（可逆圧縮）
img.save("output/tiger.png")

# BMP形式で保存します（無圧縮）
img.save("output/tiger.bmp")

# 高品質JPEGで保存します（品質95%）
img.save("output/tiger_high_quality.jpg", quality=95)

# 低品質JPEGで保存します（品質10%）
img.save("output/tiger_low_quality.jpg", quality=10)

詳細説明：

  • save()メソッドはファイル拡張子に基づいて保存形式を自動的に決定します
  • 形式ごとの特徴:
    • PNG: 可逆圧縮、透明度対応、アイコンやスクリーンショットなどに適しています
    • JPEG: 非可逆圧縮、写真に適しています、透明度には非対応
    • BMP: 無圧縮、ファイルサイズは大きいですが、すべての情報を保持します
    • GIF: アニメーション対応、最大256色
  • JPEG形式では、qualityパラメータで圧縮品質を制御できます:
    • 値の範囲は1（最低品質、最小ファイル）から95（最高品質、大きめのファイル）までです
    • デフォルト値は通常75です
  • 異なる品質設定は、画像品質とファイルサイズのバランスを取るのに役立ちます

12. 出力ディレクトリの作成

例

# 出力ディレクトリが存在することを確認します
output_dir = "output"
if not os.path.exists(output_dir):
    os.makedirs(output_dir)

詳細説明：

  • os.path.exists()指定されたディレクトリが存在するかどうかを確認します
  • os.makedirs()存在しない場合はディレクトリを作成します
  • このコードは、処理した画像を保存する場所があることを保証します
  • 生成されたすべての画像を専用のディレクトリに置くことで、プロジェクトをより整理できます
