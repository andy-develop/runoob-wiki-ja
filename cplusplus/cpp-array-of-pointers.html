C++ 指针数组

C++ 指针

ポインタ配列の概念を説明する前に、3つの整数からなる配列を使用した例を見てみましょう。

实例

#include <iostream> using namespace std; const int MAX = 3; int main () { int var[MAX] = {10, 100, 200}; for (int i = 0; i < MAX; i++) { cout << "Value of var[" << i << "] = "; cout << var[i] << endl; } return 0; }

上記のコードがコンパイルおよび実行されると、次の結果が生成されます。

Value of var[0] = 10
Value of var[1] = 100
Value of var[2] = 200

配列に int や char などのデータ型を指すポインタを格納したい場合があるかもしれません。以下は整数を指すポインタの配列の宣言です。

int *ptr[MAX];

在这里，把ptrMAX個の整数ポインタで構成される配列として宣言されています。したがって、ptrの各要素はint値を指すポインタです。次の例では3つの整数を使用しており、それらはポインタ配列に格納されます。以下に示します。

实例

#include <iostream> using namespace std; const int MAX = 3; int main () { int var[MAX] = {10, 100, 200}; int *ptr[MAX]; for (int i = 0; i < MAX; i++) { ptr[i] = &var[i]; // 赋值为整数的地址 } for (int i = 0; i < MAX; i++) { cout << "Value of var[" << i << "] = "; cout << *ptr[i] << endl; } return 0; }

上記のコードがコンパイルおよび実行されると、次の結果が生成されます。

Value of var[0] = 10
Value of var[1] = 100
Value of var[2] = 200

文字を指すポインタの配列を使って、文字列のリストを格納することもできます。次のようになります。

实例

#include <iostream> using namespace std; const int MAX = 4; int main () { const char *names[MAX] = { "Zara Ali", "Hina Ali", "Nuha Ali", "Sara Ali", }; for (int i = 0; i < MAX; i++) { cout << "Value of names[" << i << "] = "; cout << names[i] << endl; } return 0; }

上記のコードがコンパイルおよび実行されると、次の結果が生成されます。

Value of names[0] = Zara Ali
Value of names[1] = Hina Ali
Value of names[2] = Nuha Ali
Value of names[3] = Sara Ali

    char *names[MAX]はポインタ配列であり、その本質はポインタを格納する配列、つまりchar型のポインタの配列です。配列内の各要素はポインタであり、ある記憶領域を指しています。char型のアドレス：

    #include <iostream>
using namespace std;

const int MAX = 4;

int main(int argc, const char * argv[]) {

    const char *names[MAX] = {
        "Zara Ali",
        "Hina Ali",
        "Nuha Ali",
        "Sara Ali",
    };

    for (int i = 0; i < MAX; i++) {
        cout << " --- names[i]              = " << names[i] << endl;
        cout << " --- *names[i]             = " << *names[i] << endl;
        cout << endl;
        cout << " --- (*names[i] + 1)       = " << (*names[i] + 1) << endl;
        cout << " --- (char)(*names[i] + 1) = " << (char)(*names[i] + 1) << endl;
        cout << " ------------------------------------ " << endl << endl << endl << endl;
    }
    return 0;
}

    实例:

    /* -------- Pointer.cpp -------- */

#include <iostream>
using namespace std;

void pointerArray();
void pointerArray4Char();
const int MAX = 3;

int main(void){
    // pointerArray();
    pointerArray4Char();
    return 0;
}

void pointerArray(){
    int var[MAX] = {20,30,40};
    int *ptr[MAX];
    for(int i = 0; i < MAX; i++){
        ptr[i] = &var[i];//赋值为整数的地址
    }
    for(int i = 0; i < MAX; i++){
        cout << "Value of var[" << i << "] = ";
        cout << *ptr[i] <<endl;
    }
}

/**
 * 用一个指向字符的指针数组来存储一个字符串列表
 * Value of names[0] = sun;
 */
void pointerArray4Char(){
    const char *names[MAX] = {
            "sun","bin","sunbin"
    };
    for(int i = 0;i < MAX;i++){
        cout <<"Value of names[" << i << "] = ";//输出字符串的值
        cout << names[i] << endl;
        cout <<"Value of *names[" << i << "] = ";//输出指针所指向字符串首地址的值
        cout << *names[i] << endl;
        cout <<"Value of *names[" << i << "]+1 = ";//输出ascii码值
        cout << *names[i]+1 << endl;
        cout <<"Value of *(names[" << i << "]+1) = ";//输出指针所指向字符串首地址上一位的值
        cout << *(names[i]+1) << endl;
    }
}

    cout<<*names[i]<<endl;出力結果は、対応するポインタ配列の最初の文字です。なぜなら*nameは、配列の最初のアドレスを指す値を表します。

    C言語のベテランがここで伝えたいのは、C++ の演算子の優先順位は C 言語とは異なり、微妙な違いがあるということです。

    int *ptr[3];

    C++ の演算子の優先順位において、*小于[]，所以ptr先和[]結合して配列になり、次にint *結合して形成される配列の要素型はint *型を取得すると、要素がポインタである配列が得られます。これを略してポインタ配列と呼びます。

    int *(ptr[3]);

    これは上記と同じで、優先順位は次のとおりです。*小于()，() 等于[]。ptr先和[]結合して配列になり、次にint *結合して形成される配列の要素型はint *型を取得すると、要素がポインタである配列が得られます。

    int (*ptr)[3];

    これは異なります。優先順位は*小于()，()等于[]，() 和 []の優先順位は同じですが、結合の順序は左から右なので、最初に()里的* 和 ptr結合してポインタになり、次に(*ptr) 和 []と結合して配列になり、最終的にポインタとなります。ptr配列を指す、略して配列ポインタと呼ばれます。

    この定義に注意：

    const char *names[MAX] = {
    "Zara Ali",
    "Hina Ali",
    "Nuha Ali",
    "Sara Ali",
};
    • char *names[MAX]意味は次のとおりです。ポインタ配列、すなわち配列の各要素は char 型の要素を指すポインタです。
    • 通过{...}の初期化内容からわかるように、この配列の各要素には文字列が代入されています。明らかに、この要素は char 型の配列を指すポインタでなければなりません。
    • 簡単に言うと、char *names[MAX] の要素は、4 つの char 型配列を指すポインタ names[i] です。

    指针数组 与 数组指针

    #include <iostream>
using namespace std;
 
int main()
{
  cout << "数组指针--元素为指针变量的一维数组。" << endl;
  int* arr[3];      // 指针数组：元素为指针变量的一维数组。 等价于 int* (arr[3]);
  int  p[] = {1,2,3};
  for (int i = 0; i < 3; ++i)
  {
    arr[i] = &p[i];
  }
  cout << arr << endl << endl;  //指针数组的地址
  for (int i = 0; i < 3; ++i)
  {
    cout << arr+i << endl;
    cout << *(arr+i) << endl;
    cout << arr[i] << endl;
    cout << *arr[i] << endl;
  }


  cout << endl << endl << "数组指针--指向多维数组地址的指针。" << endl;
  int pp[2][2] = {
                  {1,2},
                  {3,4}
                  };    // p是个二维数组的数组名，相当于一个二级指针常量
  int (*ptr)[2] = pp;  //  数组指针：指向多维数组地址的指针
  for (int i = 0; i < 2; ++i)
  {
    cout << ptr + i << endl;  // ptr+i 是一维数组p[i]的地址，即ptr+i==&pp[i]；
    cout << &pp[i] << endl;
    cout << **(ptr + i)<< endl; // 二级指针 **(ptr + i) = pp[i][0];
    cout << pp[i][0]<< endl; 
  }
  
  cout << *(*(ptr+1)+1) << endl; // 二级指针 *(*(ptr+1)+1) = pp[1][1];
  cout << pp[1][1] << endl;
}
