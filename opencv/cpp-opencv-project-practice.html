C++ OpenCV プロジェクト実践

OpenCVプロジェクト実践は、理論知識を実際の問題に応用するプロセスです。

プロジェクト実践を通じて、OpenCVの使用スキルを強化し、現実のコンピュータビジョン問題を解決することができます。

以下は一般的なOpenCVプロジェクトの種類です：

  1. 画像処理プロジェクト：画像フィルタ、画像修復、画像強調など。

  2. 物体検出・追跡プロジェクト：顔検出、ナンバープレート認識、動体追跡など。

  3. 深層学習プロジェクト：画像分類、物体検出、セマンティックセグメンテーションなど。

  4. 動画処理プロジェクト：動画分析、リアルタイム動画処理、動画エフェクトなど。

簡単なプロジェクト

画像フィルタアプリ

画像フィルタは画像処理における基本的な操作の一つであり、異なるフィルタを適用することで、画像の外観やスタイルを変更できます。

OpenCVは、さまざまな画像フィルタ効果を実現するための豊富な関数を提供しています。

グレースケールフィルタ

グレースケールフィルタは、カラー画像をグレースケール画像に変換するプロセスです。グレースケール画像は1つのチャンネルのみを持ち、各ピクセルの値は明るさを表します。

例

#include <opencv2/opencv.hpp>

int main() {
    cv::Mat image = cv::imread("input.jpg");
    cv::Mat grayImage;
    cv::cvtColor(image, grayImage, cv::COLOR_BGR2GRAY);
    cv::imwrite("gray_output.jpg", grayImage);
    return 0;
}

ガウスぼかし

ガウスぼかしは一般的なぼかしフィルタであり、画像内のノイズを効果的に除去できます。

例

#include <opencv2/opencv.hpp>

int main() {
    cv::Mat image = cv::imread("input.jpg");
    cv::Mat blurredImage;
    cv::GaussianBlur(image, blurredImage, cv::Size(15, 15), 0);
    cv::imwrite("blurred_output.jpg", blurredImage);
    return 0;
}

以下の例は、複数のフィルタ効果（グレースケール化、ぼかし、エッジ検出など）をサポートする画像フィルタアプリを実装します。

例

#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main() {
    // 画像を読み込む
    Mat image = imread("image.jpg");
    if (image.empty()) {
        cout << "エラー：画像を読み込めませんでした。パスが正しいか確認してください。" << endl;
        return -1;
    }

    // グレースケール化
    Mat gray;
    cvtColor(image, gray, COLOR_BGR2GRAY);

    // ガウスぼかし
    Mat blurred;
    GaussianBlur(image, blurred, Size(15, 15), 0);

    // エッジ検出
    Mat edges;
    Canny(image, edges, 100, 200);

    // 結果を表示
    imshow("Original Image", image);
    imshow("Gray Image", gray);
    imshow("Blurred Image", blurred);
    imshow("Edges", edges);

    // 結果を保存
    imwrite("gray_image.jpg", gray);
    imwrite("blurred_image.jpg", blurred);
    imwrite("edges.jpg", edges);

    waitKey(0);
    return 0;
}

リアルタイム顔検出

リアルタイム顔検出は、コンピュータビジョンにおける重要な応用の一つです。

OpenCVは、画像や動画内の顔を検出するために使用できる、事前トレーニング済みのHaarカスケード分類器を提供しています。

実装手順：

  1. Haarカスケード分類器モデルを読み込みます。

  2. カメラまたは動画ファイルを開きます。

  3. 各フレームで顔検出を行います。

  4. 検出結果を描画して表示します。

例

#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main() {
    // Haarカスケード分類器を読み込む
    CascadeClassifier faceCascade;
    if (!faceCascade.load("haarcascade_frontalface_default.xml")) {
        cout << "エラー：Haarカスケード分類器を読み込めませんでした！" << endl;
        return -1;
    }

    // カメラを開く
    VideoCapture cap(0);
    if (!cap.isOpened()) {
        cout << "エラー：カメラを開けませんでした！" << endl;
        return -1;
    }

    // リアルタイムで顔を検出
    Mat frame;
    while (true) {
        cap >> frame;
        if (frame.empty()) break;

        // グレースケール画像に変換
        Mat gray;
        cvtColor(frame, gray, COLOR_BGR2GRAY);

        // 顔を検出
        vector<Rect> faces;
        faceCascade.detectMultiScale(gray, faces, 1.1, 3, 0, Size(30, 30));

        // 検出結果を描画
        for (const auto& face : faces) {
            rectangle(frame, face, Scalar(0, 255, 0), 2);
        }

        // 結果を表示
        imshow("Face Detection", frame);

        // ESCキーで終了
        if (waitKey(30) == 27) break;
    }

    // リソースを解放
    cap.release();
    destroyAllWindows();

    return 0;
}

ナンバープレート認識

ナンバープレート認識は、高度道路交通システムにおける重要な技術の一つです。

OpenCVは、ナンバープレートの検出と文字認識に使用できます。

例

#include <opencv2/opencv.hpp>

int main() {
    cv::Mat image = cv::imread("car.jpg");
    cv::Mat grayImage;
    cv::cvtColor(image, grayImage, cv::COLOR_BGR2GRAY);

    cv::CascadeClassifier plateCascade;
    plateCascade.load("haarcascade_russian_plate_number.xml");

    std::vector<cv::Rect> plates;
    plateCascade.detectMultiScale(grayImage, plates, 1.1, 2, 0 | cv::CASCADE_SCALE_IMAGE, cv::Size(30, 30));

    for (const auto& plate : plates) {
        cv::rectangle(image, plate, cv::Scalar(0, 255, 0), 2);
    }

    cv::imwrite("plate_output.jpg", image);
    return 0;
}

複雑なプロジェクト

OpenCVベースのARアプリケーション

拡張現実（AR）は、仮想情報を現実世界に重ね合わせる技術です。

OpenCVは、画像上に仮想オブジェクトを重ねるなど、簡単なARアプリの実装に使用できます。

例

#include <opencv2/opencv.hpp>

int main() {
    cv::Mat image = cv::imread("background.jpg");
    cv::Mat overlay = cv::imread("overlay.png", cv::IMREAD_UNCHANGED);

    cv::resize(overlay, overlay, cv::Size(100, 100));

    cv::Mat roi = image(cv::Rect(50, 50, overlay.cols, overlay.rows));
    overlay.copyTo(roi, overlay);

    cv::imwrite("ar_output.jpg", image);
    return 0;
}

映像監視と動体検出

映像監視と動体検出は、セキュリティ監視システムにおける重要な機能です。OpenCVは、動画内の動体を検出するために使用できます。

例

#include <opencv2/opencv.hpp>

int main() {
    cv::VideoCapture cap(0);
    if (!cap.isOpened()) {
        std::cerr << "Error opening video stream" << std::endl;
        return -1;
    }

    cv::Mat frame, prevFrame, diffFrame;
    cap >> prevFrame;
    cv::cvtColor(prevFrame, prevFrame, cv::COLOR_BGR2GRAY);

    while (true) {
        cap >> frame;
        if (frame.empty()) break;

        cv::Mat grayFrame;
        cv::cvtColor(frame, grayFrame, cv::COLOR_BGR2GRAY);

        cv::absdiff(grayFrame, prevFrame, diffFrame);
        cv::threshold(diffFrame, diffFrame, 30, 255, cv::THRESH_BINARY);

        cv::imshow("Motion Detection", diffFrame);
        if (cv::waitKey(1) == 27) break; // ESC key to exit

        prevFrame = grayFrame.clone();
    }

    cap.release();
    cv::destroyAllWindows();
    return 0;
}

複数物体追跡

複数物体追跡は、コンピュータビジョンにおける複雑な問題であり、動画内で複数のターゲットを同時に追跡します。OpenCVは、KCF、MIL、CSRTなどの複数の追跡アルゴリズムを提供しています。

例

#include <opencv2/opencv.hpp>
#include <opencv2/tracking.hpp>

int main() {
    cv::VideoCapture cap("video.mp4");
    if (!cap.isOpened()) {
        std::cerr << "Error opening video stream" << std::endl;
        return -1;
    }

    cv::Ptr<cv::MultiTracker> multiTracker = cv::MultiTracker::create();

    cv::Mat frame;
    cap >> frame;

    std::vector<cv::Rect> bboxes;
    cv::selectROIs("Tracking", frame, bboxes);

    for (const auto& bbox : bboxes) {
        multiTracker->add(cv::TrackerKCF::create(), frame, bbox);
    }

    while (true) {
        cap >> frame;
        if (frame.empty()) break;

        multiTracker->update(frame);

        for (const auto& bbox : multiTracker->getObjects()) {
            cv::rectangle(frame, bbox, cv::Scalar(255, 0, 0), 2);
        }

        cv::imshow("Multi-Object Tracking", frame);
        if (cv::waitKey(1) == 27) break; // ESC key to exit
    }

    cap.release();
    cv::destroyAllWindows();
    return 0;
}

深層学習プロジェクト：物体検出（YOLO）

YOLOモデルを使用して物体検出を実装します。

実装手順：

  1. YOLOモデルと設定ファイルを読み込みます。

  2. クラスラベルを読み込みます。

  3. 入力画像を前処理します。

  4. モデルを実行し、検出結果を解析します。

  5. 検出枠を描画して結果を表示します。

例

#include <opencv2/opencv.hpp>
#include <opencv2/dnn.hpp>
#include <iostream>

using namespace cv;
using namespace dnn;
using namespace std;

int main() {
    // YOLOモデルを読み込む
    Net net = readNet("yolov3.weights", "yolov3.cfg");
    if (net.empty()) {
        cout << "エラー：YOLOモデルを読み込めませんでした！" << endl;
        return -1;
    }

    // クラスラベルを読み込む
    ifstream classNamesFile("coco.names");
    vector<string> classNames;
    string className;
    while (getline(classNamesFile, className)) {
        classNames.push_back(className);
    }

    // 画像を読み込む
    Mat image = imread("image.jpg");
    if (image.empty()) {
        cout << "エラー：画像を読み込めませんでした。パスが正しいか確認してください。" << endl;
        return -1;
    }

    // 画像を前処理
    Mat blob = blobFromImage(image, 1 / 255.0, Size(416, 416), Scalar(0, 0, 0), true, false);
    net.setInput(blob);

    // 順伝播
    vector<Mat> outs;
    net.forward(outs, net.getUnconnectedOutLayersNames());

    // 検出結果を解析
    float confThreshold = 0.5; // 信頼度しきい値
    vector<int> classIds;
    vector<float> confidences;
    vector<Rect> boxes;
    for (const auto& output : outs) {
        for (int i = 0; i < output.rows; i++) {
            Mat scores = output.row(i).colRange(5, output.cols);
            Point classIdPoint;
            double confidence;
            minMaxLoc(scores, nullptr, &confidence, nullptr, &classIdPoint);
            if (confidence > confThreshold) {
                int centerX = static_cast<int>(output.at<float>(i, 0) * image.cols);
                int centerY = static_cast<int>(output.at<float>(i, 1) * image.rows);
                int width = static_cast<int>(output.at<float>(i, 2) * image.cols);
                int height = static_cast<int>(output.at<float>(i, 3) * image.rows);
                int left = centerX - width / 2;
                int top = centerY - height / 2;

                classIds.push_back(classIdPoint.x);
                confidences.push_back(static_cast<float>(confidence));
                boxes.push_back(Rect(left, top, width, height));
            }
        }
    }

    // 非最大値抑制
    float nmsThreshold = 0.4; // NMSしきい値
    vector<int> indices;
    NMSBoxes(boxes, confidences, confThreshold, nmsThreshold, indices);

    // 検出結果を描画
    for (int idx : indices) {
        Rect box = boxes[idx];
        rectangle(image, box, Scalar(0, 255, 0), 2);
        string label = format("%s: %.2f", classNames[classIds[idx]].c_str(), confidences[idx]);
        putText(image, label, Point(box.x, box.y - 10), FONT_HERSHEY_SIMPLEX, 0.5, Scalar(0, 255, 0), 2);
    }

    // 結果を表示
    imshow("Object Detection", image);
    waitKey(0);

    return 0;
}

動画処理プロジェクト：リアルタイム動画エフェクト

複数のエフェクト（エッジ検出、ぼかし、色反転など）をサポートするリアルタイム動画エフェクトアプリを実装します。

実装手順：

  1. カメラまたは動画ファイルを開きます。

  2. 各フレームにエフェクトを適用します。

  3. 処理後の動画を表示します。

例

#include <opencv2/opencv.hpp>
#include <iostream>

using namespace cv;
using namespace std;

int main() {
    // カメラを開く
    VideoCapture cap(0);
    if (!cap.isOpened()) {
        cout << "エラー：カメラを開けませんでした！" << endl;
        return -1;
    }

    // 動画フレームをリアルタイム処理
    Mat frame, edges;
    while (true) {
        cap >> frame;
        if (frame.empty()) break;

        // エッジ検出
        Canny(frame, edges, 100, 200);

        // 結果を表示
        imshow("Original Frame", frame);
        imshow("Edges", edges);

        // ESCキーで終了
        if (waitKey(30) == 27) break;
    }

    // リソースを解放
    cap.release();
    destroyAllWindows();

    return 0;
}
