Pythonエージェント環境設定

AIエージェントの構築を始める前に、適切な開発環境を整える必要があります。

PythonはAIおよび機械学習分野で最も人気のある言語であり、豊富なライブラリとツールを備えています。

1. Pythonバージョンの選択

AIエージェント開発には、以下を推奨します：Python 3.9以降。

以下のバージョンが一般的な選択肢です：

  • Python 3.9：安定性が高く、互換性に最も優れています
  • Python 3.10：パフォーマンスが向上し、新機能を搭載
  • Python 3.11+：最新バージョンで、パフォーマンスが最良

注意：一部のライブラリはPythonバージョンに特定の要件がある場合があります。公式ドキュメントを確認することをお勧めします。

2. Pythonのインストール

Windowsシステム

  • アクセス：Python公式サイト
  • 最新バージョンのPythonインストーラ（3.9以降）をダウンロードします
  • インストーラを実行し、必ずチェックを入れますAdd Python to PATH
  • インストール完了後、コマンドプロンプト（CMD）またはPowerShellを開き、次のように入力します：python --version、Pythonのバージョン番号が表示されます。

macOSシステム

macOSには通常Pythonがプリインストールされていますが、古いバージョンの場合があります。Homebrewを使用したインストールを推奨します：

# 安装 Homebrew（如果尚未安装）
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"

# 安装 Python
brew install python@3.9

# 验证安装
python3 --version

Linuxシステム

ほとんどのLinuxディストリビューションにはPythonがプリインストールされています。特定のバージョンが必要な場合は：

# Ubuntu/Debian
sudo apt update
sudo apt install python3.9 python3-pip

# CentOS/RHEL
sudo yum install python39 python39-pip

# 验证安装
python3 --version

3. 仮想環境の設定

仮想環境を使用すると、異なるプロジェクトの依存関係を分離し、バージョンの競合を回避できます。

仮想環境の作成

主要ライブラリの紹介

1. LangChain

LangChainは最も人気のあるAIエージェント開発フレームワークで、チェーン（Chain）およびエージェント（Agent）アプリケーションを構築するためのツールを提供します。

# LangChain 基本使用示例
from langchain.llms import OpenAI
from langchain.chat_models import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.tools import Tool

# 初始化 LLM
llm = ChatOpenAI(temperature=0.7, model="gpt-3.5-turbo")

# 创建工具
tools = [
    Tool(
        name="计算器",
        func=lambda x: str(eval(x)),
        description="用于数学计算"
    )
]

# 创建 Agent
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.ZERO_SHOT_REACT_DESCRIPTION,
    verbose=True
)

# 运行 Agent
result = agent.run("计算 25 的平方根")
print(result)

2. LlamaIndex

LlamaIndex（旧GPT Index）はドキュメントのインデックス化と検索に特化しており、ドキュメントベースのQ&Aシステムの構築に適しています。

# LlamaIndex 基本使用示例
from llama_index import VectorStoreIndex, SimpleDirectoryReader
from llama_index.llms import OpenAI

# 加载文档
documents = SimpleDirectoryReader("./data").load_data()

# 创建索引
index = VectorStoreIndex.from_documents(documents)

# 创建查询引擎
query_engine = index.as_query_engine()

# 查询
response = query_engine.query("文档中提到了哪些AI技术？")
print(response)

3. Semantic Kernel

Semantic Kernelはマイクロソフトが提供するフレームワークで、拡張性とエンタープライズ向けアプリケーションを重視しています。

# Semantic Kernel 基本使用示例
import semantic_kernel as sk
from semantic_kernel.connectors.ai.open_ai import OpenAITextCompletion

# 初始化内核
kernel = sk.Kernel()
api_key = "your-openai-api-key"
kernel.add_text_completion_service(
    "dv",
    OpenAITextCompletion("text-davinci-003", api_key)
)

# 定义技能
skill = kernel.import_semantic_skill_from_directory("./skills", "ExampleSkill")

# 运行技能
context = kernel.create_new_context()
context["input"] = "介绍一下人工智能"
result = skill["Summarize"](context)

APIキー管理

AIエージェント開発では、さまざまなAPIサービス（OpenAI、Anthropic、DeepSeekなど）へのアクセスが必要です。APIキーを安全かつ効果的に管理することが極めて重要です。

1. APIキーを安全に管理する必要があるのはなぜですか？

  • 漏洩の防止：APIキーの漏洩は経済的損失を引き起こす可能性があります
  • アクセス制御：環境ごとに異なるキーを使用
  • 設定の容易さ：チームでの協業時に設定方法を統一できます
  • 環境の分離：開発、テスト、本番環境で異なるキーを使用

使用する.envファイルで機密情報を管理し、バージョン管理システムにはコミットしません。

# config.py
import os
from pathlib import Path
from dotenv import load_dotenv

# .envファイルを読み込む
env_path = Path('.') / '.env'
load_dotenv(dotenv_path=env_path)

class Config:
    """アプリケーション設定"""

    # API Keys
    OPENAI_API_KEY = os.getenv("OPENAI_API_KEY")
    ANTHROPIC_API_KEY = os.getenv("ANTHROPIC_API_KEY")
    GOOGLE_API_KEY = os.getenv("GOOGLE_API_KEY")
    SERPAPI_API_KEY = os.getenv("SERPAPI_API_KEY")

    # アプリケーション設定
    APP_ENV = os.getenv("APP_ENV", "development")
    LOG_LEVEL = os.getenv("LOG_LEVEL", "INFO")
    DEBUG = os.getenv("DEBUG", "false").lower() == "true"

    # データベース設定
    VECTOR_DB_PATH = os.getenv("VECTOR_DB_PATH", "./data/vector_db")
    MAX_MEMORY_ITEMS = int(os.getenv("MAX_MEMORY_ITEMS", "1000"))

    @classmethod
    def validate(cls):
        """必要な設定を検証"""
        missing = []

        if not cls.OPENAI_API_KEY:
            missing.append("OPENAI_API_KEY")

        if missing:
            raise ValueError(f"必要な環境変数が不足しています: {', '.join(missing)}")

# 使用例
config = Config()
print(f"環境: {config.APP_ENV}")
print(f"デバッグモード: {config.DEBUG}")

環境設定の動的読み込み

# config_manager.py
import os
from pathlib import Path
from dotenv import load_dotenv

class ConfigManager:
    """配置管理器"""

    def __init__(self, env=None):
        self.env = env or os.getenv("APP_ENV", "development")
        self.load_config()

    def load_config(self):
        """加载配置"""
        # 先加载通用配置
        load_dotenv(dotenv_path=Path('.') / '.env')

        # 加载环境特定配置
        env_file = Path('.') / f'.env.{self.env}'
        if env_file.exists():
            load_dotenv(dotenv_path=env_file, override=True)

        # 加载本地覆盖配置（不提交到Git）
        local_file = Path('.') / f'.env.{self.env}.local'
        if local_file.exists():
            load_dotenv(dotenv_path=local_file, override=True)

    def get(self, key, default=None):
        """获取配置值"""
        return os.getenv(key, default)

# 使用示例
config = ConfigManager(env="development")
print(f"当前环境: {config.get('APP_ENV')}")
print(f"API Key: {config.get('OPENAI_API_KEY')[:10]}...")  # 只显示前10个字符

5. キーのローテーションと監視

本番環境では、キーのローテーションと監視の実施を推奨します：

# key_manager.py
import os
import time
from datetime import datetime, timedelta

class APIKeyManager:
    """API Key 管理器"""

    def __init__(self):
        self.keys = {}
        self.key_history = []
        self.load_keys()

    def load_keys(self):
        """加载 API Keys"""
        # 可以从环境变量、数据库或密钥管理服务加载
        self.keys = {
            "openai": {
                "current": os.getenv("OPENAI_API_KEY"),
                "previous": os.getenv("OPENAI_API_KEY_PREVIOUS"),
                "created_at": datetime.now(),
                "rotation_days": 30  # 30天轮换一次
            },
            "anthropic": {
                "current": os.getenv("ANTHROPIC_API_KEY"),
                "previous": None,
                "created_at": datetime.now(),
                "rotation_days": 30
            }
        }

    def get_key(self, service):
        """获取当前可用的 Key"""
        key_info = self.keys.get(service)
        if not key_info:
            raise ValueError(f"未配置服务 {service} 的 API Key")

        # 检查是否需要轮换
        if self.should_rotate(key_info):
            self.rotate_key(service)

        return key_info["current"]

    def should_rotate(self, key_info):
        """检查是否需要轮换 Key"""
        rotation_days = key_info.get("rotation_days", 30)
        created_at = key_info.get("created_at", datetime.now())

        age = datetime.now() - created_at
        return age.days >= rotation_days

    def rotate_key(self, service):
        """轮换 API Key"""
        # 在实际应用中，这里会从密钥管理服务获取新 Key
        print(f"轮换 {service} 的 API Key...")

        key_info = self.keys[service]
        key_info["previous"] = key_info["current"]
        # 这里应该从安全的地方获取新 Key
        # key_info["current"] = get_new_key_from_vault(service)
        key_info["created_at"] = datetime.now()

        # 记录历史
        self.key_history.append({
            "service": service,
            "rotated_at": datetime.now(),
            "old_key": key_info["previous"][:10] + "..." if key_info["previous"] else None
        })

# 使用示例
key_manager = APIKeyManager()
openai_key = key_manager.get_key("openai")
print(f"OpenAI Key: {openai_key[:10]}...")

4. テストツール

pytest

# tests/test_agent.py
import pytest
from src.agents.base_agent import BaseAgent

def test_agent_initialization():
    """测试 Agent 初始化"""
    agent = BaseAgent(name="测试Agent")
    assert agent.name == "测试Agent"
    assert agent.messages == []

def test_agent_add_message():
    """测试添加消息"""
    agent = BaseAgent()
    agent.add_message("user", "你好")

    assert len(agent.messages) == 1
    assert agent.messages[0]["role"] == "user"
    assert agent.messages[0]["content"] == "你好"

# 运行所有测试
pytest

pytest tests/test_agent.py

pytest tests/test_agent.py::test_agent_initialization

pytest -v

pytest --cov=src

      - id: check-added-large-files

    rev: 23.12.1

loguru（ログライブラリ）

from loguru import logger
import sys

# 配置日志
logger.remove()  # 移除默认处理器
logger.add(
    sys.stderr,
    format="<green>{time:YYYY-MM-DD HH:mm:ss}</green> | <level>{level: <8}</level> | <cyan>{name}</cyan>:<cyan>{function}</cyan>:<cyan>{line}</cyan> - <level>{message}</level>",
    level="INFO"
)

logger.add(
    "logs/app_{time}.log",
    rotation="500 MB",  # 每500MB轮转
    retention="30 days",  # 保留30天
    compression="zip",  # 压缩旧日志
    level="DEBUG"
)

# 使用日志
logger.info("应用启动")
logger.debug(f"配置加载: {config}")
logger.error("API调用失败", exc_info=True)
