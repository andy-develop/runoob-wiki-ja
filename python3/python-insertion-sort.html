挿入ソート（英語：Insertion Sort）は、シンプルで直感的なソートアルゴリズムです。その動作原理は、整列済みの列を構築し、未整列のデータについて、整列済みの列を後ろから前に向かって走査し、対応する位置を見つけて挿入するというものです。

サンプル

def insertionSort(arr): for i in range(1, len(arr)): key = arr[i] j = i-1 while j >=0 and key < arr[j] : arr[j+1] = arr[j] j -= 1 arr[j+1] = key arr = [12, 11, 13, 5, 6] insertionSort(arr) print ("ソート後の配列:") for i in range(len(arr)): print ("%d" %arr[i])

# 一次往数组添加多个数字
def AppendNumbers(array):
    num = input('Numbers:(split by spaces)\t').split()
    for i in num:
        array.append(int(i))
    print('排序前数组：{}.'.format(array))

def InsertionSort(array):
    AppendNumbers(array)  # 添加

    list = []
    while True:
        for i in array:
            minimum = min(array)
            if i == minimum:
                list.append(i)
                array.remove(i)  # 删去最小值

        if array == []:
            break

    print('排序后数组：{}.'.format(list))

def insertionSort(arr):
    #从要排序的列表第二个元素开始比较
    for i in range(1,len(arr)):
        j = i
        #从大到小比较，直到比较到第一个元素
        while j > 0:
            if arr[j] < arr[j-1]:
                arr[j-1],arr[j] = arr[j],arr[j-1]
            j -= 1        
    return arr
print(insertionSort(arr))

    # 插入排序算法实现
def insertionSort(list):
    n = len(list)
    for i in range(1,n):  #默认数列中的第一个元素已排好，故从第二个元素开始插入       
        j = i
        while list[j] < list[j-1] and j>0:  #和已经排好序的数从右往左比较，小于则交换位置            
            list[j],list[j-1] = list[j-1],list[j]
            j-=1    
     return list
