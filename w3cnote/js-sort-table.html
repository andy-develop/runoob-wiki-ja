JavaScript/CSS テーブルソート機能

カテゴリ

この章では、JS/CSSを使用してテーブルソート機能を実装します。

まず効果は以下の通りです。ボタンをクリックしてソートを実行します（Nameフィールドに基づいてアルファベット昇順でソート）：

ソートコード

例

function sortTable() { var table, rows, switching, i, x, y, shouldSwitch; table = document.getElementById("myTable"); switching = true; /*ループ文を設定する*/ while (switching) { //ループ終了フラグを設定する switching = false; rows = table.rows; /*テーブルの行をループする*/ for (i = 1; i < (rows.length - 1); i++) { //要素を交換するかどうかを設定する shouldSwitch = false; /*比較する要素を取得する*/ x = rows[i].getElementsByTagName("TD")[0]; y = rows[i + 1].getElementsByTagName("TD")[0]; //次の要素を現在の要素と入れ替えるかどうかを判断する if (x.innerHTML.toLowerCase() > y.innerHTML.toLowerCase()) { //要素交換フラグを設定し、現在のループを終了する shouldSwitch = true; break; } } if (shouldSwitch) { /*要素の入れ替えフラグが true の場合、交換操作を実行する*/ rows[i].parentNode.insertBefore(rows[i + 1], rows[i]); switching = true; } } }

表ヘッダーをクリックしてソートを実現

表ヘッダーのフィールドをクリックして、テーブルをソートします。

最初のクリックでは、ソート方向は昇順（A から Z）です。

もう一度クリックすると、ソート方向は降順（Z から A）になります。

例

function sortTable(n) { var table, rows, switching, i, x, y, shouldSwitch, dir, switchcount = 0; table = document.getElementById("myTable2"); switching = true; //昇順に設定する dir = "asc"; /*ループ文を設定する*/ while (switching) { //ループ終了フラグを設定する switching = false; rows = table.rows; /*リストをループする、2行目から開始*/ for (i = 1; i < (rows.length - 1); i++) { //要素を交換するかどうかを設定する shouldSwitch = false; /*比較する要素を取得する、 one from current row and one from the next:*/ x = rows[i].getElementsByTagName("TD")[n]; y = rows[i + 1].getElementsByTagName("TD")[n]; /*次の要素を現在の要素と入れ替えるかどうかを判断する（asc または desc）:*/ if (dir == "asc") { if (x.innerHTML.toLowerCase() > y.innerHTML.toLowerCase()) { //要素交換フラグを設定し、現在のループを終了する shouldSwitch = true; break; } } else if (dir == "desc") { if (x.innerHTML.toLowerCase() < y.innerHTML.toLowerCase()) { //要素交換フラグを設定し、現在のループを終了する shouldSwitch = true; break; } } } if (shouldSwitch) { /*要素の入れ替えフラグが true の場合、交換操作を実行する*/ rows[i].parentNode.insertBefore(rows[i + 1], rows[i]); switching = true; //交換が完了するたびに、switchcount を 1 増やします switchcount ++; } else { /*すべての要素のソートが完了し、direction が "asc" に設定されている場合、direction を "desc" に設定して、ループを再度実行します*/ if (switchcount == 0 && dir == "asc") { dir = "desc"; switching = true; } } } }
