JavaScript/CSS リストソート機能

カテゴリ プログラミング技術

この章では、JS/CSSを使用してリストソート機能を実装する方法を学びます。

まず、以下のような効果をご覧ください。ボタンをクリックすると並べ替えが実行されます：

基本HTMLコード

例

<ul id="id01"> <li>Runoob</li> <li>Google</li> <li>Taobao</li> <li>Zhihu</li> <li>Baidu</li> <li>Weibo</li> </ul> <script>
function sortList() { var list, i, switching, b, shouldSwitch; list = document.getElementById("id01"); switching = true; /*ループ文を設定する*/ while (switching) { //ループ終了フラグを設定する switching = false; b = list.getElementsByTagName("LI"); //リストをループする for (i = 0; i < (b.length - 1); i++) { //要素を入れ替えるかどうかを設定する shouldSwitch = false; /*次の要素を現在の要素と入れ替えるかどうかを判断する*/ if (b[i].innerHTML.toLowerCase() > b[i + 1].innerHTML.toLowerCase()) { /*2つの要素をアルファベット順に比較し、次の要素のアルファベットが現在の要素よりも小さい場合、要素入れ替えフラグを設定し、現在のループを終了する*/ shouldSwitch = true; break; } } if (shouldSwitch) { /*要素の入れ替え位置が true に設定されている場合、交換操作を実行する*/ b[i].parentNode.insertBefore(b[i + 1], b[i]); switching = true; } } }
</script>

昇順または降順の並べ替え

最初にボタンをクリックすると、並べ替え方向は昇順（A から Z）になります。

再度クリックすると、並べ替え方向は降順（Z から A）になります。

まず、以下のような効果をご覧ください。ボタンをクリックすると並べ替えが実行されます：

例

<ul id="id01"> <li>Runoob</li> <li>Google</li> <li>Taobao</li> <li>Zhihu</li> <li>Baidu</li> <li>Weibo</li> </ul> <script>
function sortListDir() { var list, i, switching, b, shouldSwitch, dir, switchcount = 0; list = document.getElementById("id01"); switching = true; //昇順の並べ替えを設定する dir = "asc"; //ループ文を設定する while (switching) { //ループ終了フラグを設定する switching = false; b = list.getElementsByTagName("LI"); //リストをループする for (i = 0; i < (b.length - 1); i++) { //要素を入れ替えるかどうかを設定する shouldSwitch = false; /*次の要素を現在の要素と入れ替えるかどうかを判断する（asc または desc）：*/ if (dir == "asc") { if (b[i].innerHTML.toLowerCase() > b[i + 1].innerHTML.toLowerCase()) { /*2つの要素をアルファベット順に比較し、次の要素のアルファベットが現在の要素よりも小さい場合、要素入れ替えフラグを設定し、現在のループを終了する*/ shouldSwitch = true; break; } } else if (dir == "desc") { if (b[i].innerHTML.toLowerCase() < b[i + 1].innerHTML.toLowerCase()) { /*2つの要素をアルファベット順に比較し、次の要素のアルファベットが現在の要素よりも小さい場合、要素入れ替えフラグを設定し、現在のループを終了する*/ shouldSwitch= true; break; } } } if (shouldSwitch) { /*要素の入れ替え位置が true に設定されている場合、交換操作を実行する*/ b[i].parentNode.insertBefore(b[i + 1], b[i]); switching = true; //交換が完了するたびに、switchcount を 1 増やす switchcount ++; } else { /*すべての要素の並べ替えが完了し、direction が "asc" に設定されている場合、direction を "desc" に設定して、再びループを実行する*/ if (switchcount == 0 && dir == "asc") { dir = "desc"; switching = true; } } } }
</script>
