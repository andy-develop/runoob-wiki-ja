4.1 ES6関数

分類 ES6チュートリアル

関数パラメータの拡張

デフォルトパラメータ

基本的な使い方

function fn(name,age=17){ console.log(name+","+age); } fn("Amy",18); // Amy,18 fn("Amy",""); // Amy, fn("Amy"); // Amy,17

注意点：関数のデフォルトパラメータを使用する場合、同名のパラメータは許可されません。

//エラーにならない function fn(name,name){ console.log(name); } //エラーになる //SyntaxError: Duplicate parameter name not allowed in this context function fn(name,name,age=17){ console.log(name+","+age); }

デフォルトパラメータは、パラメータが渡されない場合、またはパラメータが undefined の場合にのみ使用されます。null は有効な値として渡されたとみなされます。

function fn(name,age=17){ console.log(name+","+age); } fn("Amy",null); // Amy,null

関数パラメータのデフォルト値には一時的デッドゾーンが存在します。関数パラメータのデフォルト値の式では、まだ初期化・代入されていないパラメータの値を他のパラメータのデフォルト値として使用することはできません。

function f(x,y=x){ console.log(x,y); } f(1); // 1 1 function f(x=y){ console.log(x); } f(); // ReferenceError: y is not defined

残余引数

残余引数は、引数の個数が不確定な場合に使用します。形式は ...変数名 で、... に名前付きパラメータ識別子を加えたものです。名前付きパラメータは引数リストの最後に配置し、かつ残余引数は1つだけです。

基本的な使い方

function f(...values){ console.log(values.length); } f(1,2); //2 f(1,2,3,4); //4

アロー関数

アロー関数は、より簡潔な関数の記述方法を提供します。基本構文は次のとおりです。

パラメータ => 関数本体

基本的な使い方：

var f = v => v; //と等価です var f = function(a){ return a; } f(1); //1

アロー関数にパラメータがない場合、または複数のパラメータがある場合は、()括弧で囲む必要があります。

var f = (a,b) => a+b; f(6,2); //8

アロー関数の関数本体に複数行のステートメントがある場合は、{}中括弧で囲んでコードブロックを表します。1行のステートメントのみで、結果を返す必要がある場合は、中括弧を省略できます。{}、結果は自動的に返されます。

var f = (a,b) => { let result = a+b; return result; } f(6,2); // 8

アロー関数がオブジェクトを返す場合、コードブロックと区別するために、()オブジェクトを括弧で囲む必要があります。

//エラーになる var f = (id,name) => {id: id, name: name}; f(6,2); // SyntaxError: Unexpected token : //エラーにならない var f = (id,name) => ({id: id, name: name}); f(6,2); // {id: 6, name: 2}

注意点：this、super、arguments、new.target のバインドがありません。

var func = () => { //アロー関数内には this オブジェクトがありません。 //このときの this は外側の this オブジェクト、つまり Window です。 console.log(this) } func(55) // Window var func = () => { console.log(arguments) } func(55); // ReferenceError: arguments is not defined

アロー関数の本体における this オブジェクトは、関数を定義した時点のオブジェクトであり、関数を使用した時点のオブジェクトではありません。

function fn(){ setTimeout(()=>{ //定義時、this は fn 内の this オブジェクトにバインドされます。 console.log(this.a); },0) } var a = 20; //fn の this オブジェクトは {a: 18} です。 fn.call({a: 18}); // 18

コンストラクタとして使用できません。つまり new コマンドを使用できません。使用するとエラーになります。

使用に適したシーン

ES6 以前、JavaScript の this オブジェクトは頭を悩ませる存在でした。コールバック関数では、外部の this をコールバック関数に渡すために、var self = this のようなコードをよく見かけました。しかし、アロー関数があればその必要はなく、直接 this を使用できます。

//コールバック関数 var Person = { 'age': 18, 'sayHello': function () { setTimeout(function () { console.log(this.age); }); } }; var age = 20; Person.sayHello(); // 20 var Person1 = { 'age': 18, 'sayHello': function () { setTimeout(()=>{ console.log(this.age); }); } }; var age = 20; Person1.sayHello(); // 18

したがって、this コンテキストを維持する必要がある場合には、アロー関数を使用できます。

使用に適さないシーン

メソッドを定義する場合で、そのメソッドが this を含む場合

var Person = { 'age': 18, 'sayHello': ()=>{ console.log(this.age); } }; var age = 20; Person.sayHello(); // 20 //このとき、this はグローバルオブジェクトを指します。 var Person1 = { 'age': 18, 'sayHello': function () { console.log(this.age); } }; var age = 20; Person1.sayHello(); // 18 //このときの this は Person1 オブジェクトを指します。

動的な this が必要な場合

var button = document.getElementById('userClick'); button.addEventListener('click', () => { this.classList.toggle('on'); });

button のリスナー関数がアロー関数であるため、リスナー関数内の this は定義時の外側の this オブジェクト、つまり Window を指し、クリックされたボタンオブジェクトを操作できません。
