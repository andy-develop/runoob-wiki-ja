C言語の定番100例

問題：ある数がその因子の和とちょうど等しい場合、この数を「完全数」と呼びます。例えば6=1＋2＋3、プログラミングで1000以内のすべての完全数を見つけてください。

プログラム分析：参照：C 練習例14。

例

// Created by www.runoob.com on 15/11/9. // Copyright © 2015年 初心者向けチュートリアル. All rights reserved. // #include <stdio.h> int main() { int i, j, sum; printf("1000以内の完全数は次のとおりです：\n"); for (i = 1; i <= 1000; i++) { sum = 0; // ループごとにsumをリセット // iのすべての真因子を求め、合計する for (j = 1; j < i; j++) { if (i % j == 0) { sum += j; } } // 合計がi自身と等しければ、完全数である if (sum == i) { printf("%d = 1", i); // すべての完全数には1が含まれる // 他の因子を出力 for (j = 2; j < i; j++) { if (i % j == 0) { printf(" + %d", j); } } printf("\n"); } } return 0; }

上記の例の出力結果は次のとおりです：

6 = 1 + 2 + 3
28 = 1 + 2 + 4 + 7 + 14
496 = 1 + 2 + 4 + 8 + 16 + 31 + 62 + 124 + 248

    参考答案の反復回数は次の結論を利用しています：正の整数Nがあるとき、N/2 < a < N の場合、a は n の因子にはなり得ません。なぜなら 1<N/a<2 が導かれるため、反復範囲は i<=N/2 に縮小されます。

    上記の状況は一般化できます。正の整数Nがあり、Nのすべての因子を a0 a1 ... an とし、a0<a1<...<an とします。このとき次の結論が成り立ちます：a1 が N の因子なら、N/a1 も N の因子であり、N=a0*an=a1*an-1=... となります。an-1<i<an の範囲では、i は N の因子になり得ません。なぜなら a0=N/an<N/i<N/an-1=a1 が導かれ、a0 と a1 の範囲には N の因子が存在しないからです。したがって、a が N の因子であると確認できれば、N のもう一つの因子 N/a を確定でき、反復範囲を i<N/a に縮小できます。

    #include <stdio.h>

int perfect(int n)
{
    int s=1, t=n, i;
    for(i=2; i<t; i++)
    {
        if(n % i == 0)
        {
            t = n / i;
            s += i;
            if(t != i) s += t;
        }
    }
    if(n==s) return 1;
    else return 0;
}

int main()
{
    int i, j, t;
    for(i=1; i<1001; i++)
    {
        if(perfect(i))
        {
            printf("%d=1", i);
            t = i;
            for(j=2; j<t; j++)
            {
                if(i % j == 0)
                {
                    t = i / j;
                    printf("+%d+%d", j, t);
                }
            }
            printf("\n");
        }
    }
}
