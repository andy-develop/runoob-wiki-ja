問題：文字列の反転、例えば文字列 "www.runoob.com" を "moc.boonur.www" に反転します。

プログラム分析：なし。

例

// Created by www.runoob.com on 15/11/9. // Copyright © 2015年 菜鸟教程. All rights reserved. // #include <stdio.h> void reverse(char* s) { // 文字列の長さを取得 int len = 0; char* p = s; while (*p != 0) { len++; p++; } // 交換 ... int i = 0; char c; while (i <= len / 2 - 1) { c = *(s + i); *(s + i) = *(s + len - 1 - i); *(s + len - 1 - i) = c; i++; } } int main() { char s[] = "www.runoob.com"; printf("'%s' =>\n", s); reverse(s); // 文字列を反転 printf("'%s'\n", s); return 0; }

上記の例の出力結果は次のとおりです：

'www.runoob.com' =>
'moc.boonur.www'

    #include<stdio.h>
#include<string.h>

void reverse(char *str)
{
    int len;
    char tmp;
    char *begin,*end;
    len=strlen(str); //获取字符串长度]
    begin=str;       //指针begin指向字符串首地址
    end=str+len-1;     //指针end指向字符串尾地址
    while(begin<end)
    {
        tmp=*begin;
        *begin=*end;
        *end=tmp;
        begin++;
        end--;
    }
}

int main()
{
    char string[]="www.runoob.com";
    reverse(string);
    puts(string);
    return 0;
}

void reverse(char *str)
{
    char *p = str, strNew[100];
    int i = 0;
    while (str[i] != '\n') {
        strNew[i] = str[i];
        ++i;
     }
    strNew[i] = '\0';//把原字符串末尾置'\0'
    while (*p != '\n') *(p++) = strNew[--i];//把原字符逆序
    *p = '\0';//逆序后末尾置'\0'
}
