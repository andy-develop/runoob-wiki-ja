問題：文字列内の指定された文字を削除します。例：文字列 "aca" から a 文字を削除します。

プログラム分析：なし。

例

// Created by www.runoob.com on 15/11/9. // Copyright © 2015年 菜鸟教程. All rights reserved. // #include <stdio.h> #include <stdlib.h> #include <string.h> // 文字列内の指定された文字を削除する関数 char* deleteCharacters(char* str, char* charSet) { // 削除する文字を格納するためのハッシュテーブル int hash[256] = {0}; // サイズ256の配列を使用してASCII文字を格納し、0に初期化する // 削除する文字セットが空の場合は、元の文字列をそのまま返す if (charSet == NULL) return str; // 削除する文字をマークする for (int i = 0; i < strlen(charSet); i++) hash[(unsigned char)charSet[i]] = 1; // 削除する文字のASCIIコード位置を1にマークする int currentIndex = 0; // 現在の文字列インデックスを追跡するために使用する // 文字列を走査して指定された文字を削除する for (int i = 0; i < strlen(str); i++) { // 現在の文字が削除する文字セットに含まれていない場合は、その文字を保持する if (!hash[(unsigned char)str[i]]) str[currentIndex++] = str[i]; } str[currentIndex] = '\0'; // 文字列の末尾に空文字を追加して、文字列の終わりを示す return str; } int main() { char s[] = "a"; // 削除する文字 char s2[] = "aca"; // 対象の文字列 printf("%s\n", deleteCharacters(s2, s)); // 指定された文字を削除した後の文字列を出力する return 0; }

    len=len1-cnt;   //新字符串的长度
