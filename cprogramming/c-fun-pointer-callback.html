関数ポインタ

関数ポインタは関数を指すポインタ変数です。

通常、私たちが言うポインタ変数は整数型、文字型、配列などの変数を指しますが、関数ポインタは関数を指します。

関数ポインタは通常の関数と同じように、関数の呼び出しや引数の受け渡しに使用できます。

関数ポインタ型の宣言：

typedef int (*fun_ptr)(int,int); // 声明一个指向同样参数、返回值的函数指针类型

例

次の例では、関数 max を指す関数ポインタ変数 p を宣言しています。

例

#include <stdio.h> int max(int x, int y) { return x > y ? x : y; } int main(void) { /*p は関数ポインタです*/ int (* p)(int, int) = & max; // & は省略可能 int a, b, c, d; printf("3つの数字を入力してください:"); scanf("%d %d %d", & a, & b, & c); /*関数を直接呼び出すのと同等で、d = max(max(a, b), c) です。*/ d = p(p(a, b), c); printf("最大の数字は: %d\n", d); return 0; }

コンパイルして実行すると、出力結果は次のようになります：

请输入三个数字:1 2 3
最大的数字是: 3

コールバック関数

関数ポインタをある関数のパラメータとして使う

関数ポインタ変数はある関数のパラメータとして使用できます。コールバック関数とは、関数ポインタを介して呼び出される関数のことです。

簡単に言うと、コールバック関数は、他の関数が実行されるときに呼び出される、あなたが実装した関数です。

以下は知乎（Zhihu）の著者・常溪玲氏による解説です：

あなたが店に買い物に行き、ちょうど欲しい商品が在庫切れだったため、店員に電話番号を預けました。数日後、店に在庫が入ると、店員があなたに電話をかけ、あなたはその電話を受けて店に商品を取りに行きました。この例では、あなたの電話番号はコールバック関数、店員に電話番号を預けることはコールバック関数の登録、店に在庫が入ることはコールバックに関連するイベントの発生、店員が電話をかけることはコールバック関数の呼び出し、店に商品を取りに行くことはコールバックイベントへの応答と呼ばれます。

例

例の中でpopulate_array()この関数は3つのパラメータを定義しており、その3番目のパラメータは関数ポインタで、この関数を介して配列の値を設定します。

例ではコールバック関数を定義していますgetNextRandomValue()、これはランダムな値を返し、関数ポインタとして渡されますpopulate_array()関数。

populate_array()呼び出し、10回コールバック関数を呼び出し、コールバック関数の戻り値を配列に代入します。

例

#include <stdlib.h> #include <stdio.h> void populate_array(int *array, size_t arraySize, int (*getNextValue)(void)) { for (size_t i=0; i<arraySize; i++) array[i] = getNextValue(); } // ランダム値を取得 int getNextRandomValue(void) { return rand(); } int main(void) { int myarray[10]; /*getNextRandomValue に括弧を付けることはできません。付けないとコンパイルできません。括弧を付けると、このパラメータを渡すときに int が渡されることになり、関数ポインタではなくなるからです。*/ populate_array(myarray, 10, getNextRandomValue); for(int i = 0; i < 10; i++) { printf("%d ", myarray[i]); } printf("\n"); return 0; }

コンパイルして実行すると、出力結果は次のようになります：

16807 282475249 1622650073 984943658 1144108930 470211272 101027544 1457850878 1458777923 2007237709

    より標準的な書き方なら GCC でそのまま通ります。ポインタ変数はポインタ変数として扱います。

    #include <stdlib.h>
#include <stdio.h>

// 回调函数
void populate_array(int *array, int arraySize, int (*getNextValue)(void))
{
    int i;
    for(i=0; i<arraySize; i++){
        *(array+i) = getNextValue();
    }
}

// 获取随机值
int getNextRandomValue(void)
{
    return rand();
}

int main(void)
{
    int myarray[10];
    int i;
    populate_array(myarray, 10, getNextRandomValue);
    for(i=0; i<10; i++) {
        printf("%d ", myarray[i]);
    }
    printf("\n");
    return 0;
}

    コールバック関数についての考察

    A 関数内でパラメータと C 関数を B 関数に渡し、B 関数が C 関数を呼び出します。B 関数が呼び出す動作をコールバックと呼び、C 関数をコールバック関数と呼びます。

    例を挙げる

    #include <stdio.h>

// 回调函数a
void callback_a(int var_a){    
    printf("Call callback_a, var: %d\n\n", var_a);
}

// 回调函数b
void callback_b(int var_b){    
    printf("Call callback_b, var: %d\n\n", var_b);
}

// 回调动作
void callback_act(int x, void (*callback) (int var)){    
    printf("Call callback_act, var_x: %d\n\n", x);    
    callback(x);
}


void main(){    
    int a = 1, b = 2;    
    callback_act(a, callback_a);    
    callback_act(b, callback_b);    
    printf("Main program has done.");
}

    実行結果：

    Call callback_act, var_x: 1
Call callback_a, var: 1
Call callback_act, var_x: 2
Call callback_b, var: 2
Main program has done.

    よく考えると、コールバック関数の役割は通常の関数でも十分に達成できます。実際、コールバック関数の最大の意義は疎結合にあり、コード間の結合度を下げることです。

    詳細は：C 言語コールバック関数の詳しい解説

    size_t について、vcruntime.h で次のような定義を見つけました：

    // Definitions of common types
#ifdef _WIN64
    typedef unsigned __int64 size_t;
    typedef __int64          ptrdiff_t;
    typedef __int64          intptr_t;
#else
    typedef unsigned int     size_t;
    typedef int              ptrdiff_t;
    typedef int              intptr_t;
#endif

    つまり、マシンが 64 ビットかどうかを判断し、そうであれば：

    typedef unsigned __int64 size_t;

    そうでなければ：

    typedef unsigned int     size_t;

    まとめると、システムのビット数に応じて、対応する 64 ビット int か 32 ビット int になるということです。

    メモリにはさまざまな変数が格納されるだけでなく、関数もコンパイル時にメモリに格納されます。したがって、関数ポインタは関数を指すポインタです。

    関数ポインタを使うとコールバック関数の操作を実現できます。以下によく使われる関数ポインタの操作を挙げます：

    // 函数指针示例 
#include <stdio.h>
#include <stdlib.h>
// 几个示例函数 
void print() { printf("哈哈哈\n"); } 
void print1() { printf("哈哈哈哈哈哈\n"); } 
int sum(int a, int b) { return a + b; } 
void printInt(int a) { printf("%d ", a); } 
int *pointer() { 
    int *a = (int *)malloc(sizeof(int)); 
    *a = 11; return a; 
} 
// 把函数指针作为函数参数传递（回调函数） 
// 函数声明（注意类型的表示） 
void traverse(int[], void (*)(int)); 
/// @brief 遍历数组元素（回调函数例子） 
/// @param array 数组 
/// @param callback 回调函数，无返回值，一个int参数 
void traverse(int array[], void (*callback)(int)) { 
    // 遍历前面3个用回调函数访问 
    int i; 
    for (i = 0; i < 3; i++) { 
        callback(array[i]); 
    } 
    printf("\n"); 
} 
// 函数返回函数指针 
// 返回一个返回值为int，带两个int参数的函数指针
// 函数本身名为fun，本身无参数 
int (*fun())(int, int) { return sum; } 
// 函数返回函数指针，带参数型 
// 返回一个返回值为void，不带参数的函数指针 
// 函数本身名为fun1，带一个int参数 
void (*fun1(int a))() { if (a == 1) { return print1; } return print; } 
int main() { 
    // 函数指针p：无参，返回类型void 
    // 让函数指针p指向print函数 
    void (*p)() = print; 
    //执行p 
    p(); 
    // 函数指针sp：两个int参数，返回类型int 
    // 让函数指针sp指向sum函数 
    int (*sp)(int, int); 
    // 执行sp 
    printf("%d\n", sum(1, 2)); 
    // 一个返回类型为指针的函数指针 
    // 把pointer赋值给函数指针ip 
    int *(*ip)() = pointer; printf("%d\n", *ip()); 
    // 一个长度为2的函数指针数组fp[2]，其中函数指针是void无参函数 
    void (*fp[2])(); 
    // 给函数指针数组中元素赋值 
    fp[0] = print; fp[1] = print1; 
    // 执行它们 
    fp[0](); fp[1](); 
    // 使用回调函数 
    int arr[] = {1, 2, 3}; traverse(arr, printInt); 
    // 执行返回函数指针的函数得到函数指针，将函数返回值赋值给函数指针getSum 
    int (*getSum)(int, int) = fun(); 
    printf("%d\n", getSum(3, 4)); 
    // 传参的返回函数指针的函数，下面两个分别赋值给函数指针pp1，pp2 
    void (*pp1)() = fun1(1); void (*pp2)() = fun1(0); 
    // 执行 
    pp1(); 
    pp2(); 
    // 输出函数指针类型变量的大小，可见使用"返回值 (*)(参数)"可以表示一个函数指针类型的指针变量 
    printf("%d %d %d\n", sizeof(void (*)()), sizeof(void (*)(int)), sizeof(int (*)())); return 0; 
}

    @5階と@8階について考察し修正した個人的な見解は次のとおりです：

    #include <stdio.h>

int sum(int num1, int num2){
    return num1 + num2;
}

//sum2调用sum函数
int sum2(int num3, int (*shapePara)(int, int), int num4, int num5){
    /*
        shapePara形式--函数参数名，名字不重要（与sum函数一致，亦可），
        实际的指向关系是shapePara --> callback_sum(可省,详见下面多行注释) --> sum,调用sum函数里面的功能;
    */
    return  num3 * shapePara(num4, num5);
}

int main()
{
    printf("SUM = %d\n", sum(1, 2));
    printf("SUM2(2倍的SUM) = %d\n", sum2(2, sum, 1, 2));
    /*
        函数指针的建立，以便回调使用
        int (*callback_sum)(int, int) = sum;（可省略，写成上面形式）
        printf("SUM2(2倍的SUM) = %d\n", sum2(2, callback_sum, 1, 2));  (功能相同)

        其次上面只用函数名sum，是作为函数指针，以便减少代码量。
    */
    return 0;
}
/*
    此外多提一嘴，以上代码采用顺序编写，所以无需进行事先的函数声明；
    否则，要将main函数后面的函数统统进行事先声明，不然会报错的；
*/
