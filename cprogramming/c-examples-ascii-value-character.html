C言語例 - 文字をASCIIコードに変換

C言語例

ASCIIは128個の文字を定義しています。

分類：

  • 一：0-31、127（削除キー）は制御文字です
  • 二：空白文字：スペース（32）、タブ、垂直タブ、改行、復帰。
  • 三：表示可能な文字：a-z、A-Z、0-9、~、！、@、#、%、^、&、*、（、）、-、+、{、}、[、]、'、"、<、>、，、？、/、|、\、_、：、;、.、および読点（、）。

例

#include <stdio.h> int main() { char c; printf("文字を入力してください："); // ユーザー入力を読み取る scanf("%c", &c); // %d で整数を表示する // %c は対応する文字を表示する printf("%c の ASCII は %d です", c, c); return 0; }

    この例では、いくつかの変更を加えることで、長い文字列を一度に読み取り、ASCIIコードを1つずつ表示できます。

    #include <stdio.h>
#define MAX_STRING_LENGTH 65535 // 最大字符串长度

int main(){
  char s[MAX_STRING_LENGTH];
  printf("请输入长度小于 %d 的任意字符：",MAX_STRING_LENGTH);
  scanf("%s",s);    // 读取字符串。
  for(int i = 0; s[i]; i++){
    printf("%c的ASCII:%d\t",s[i],s[i]);
  }
}

    文字から対応するASCIIを調べるなら、ASCIIを文字に変換するものも書いてみてはどうですか？相互に変換できたら良くないですか？

    以下は改善したコードです。初心者なので、お手柔らかにお願いします。ご指導よろしくお願いします。ありがとうございます。

    #include <stdio.h>
#define MAX_ASCII 127

int main()
{
    char num,enter;
    int temp=1;
    for(;temp>0;)
    {
        printf("----------------------------\n");
        printf("|**      开始            **|\n");
        printf("|**ASCII  转  字符  按:1 **|\n");
        printf("|**字符   转  ASCII 按:2 **|\n");
        printf("|**      结束       按:0 **|\n");
        printf("----------------------------\n");
        scanf("%d",&temp);
        if(temp==1)
        {
            printf("请输入数值小于 %d 的任意字符：",MAX_ASCII);
            scanf("%d",&num);
            printf("ASCII为 %d ,对应的字符为 %c \n",num,num);
        }
        if(temp==2)
        {
            printf("输入一个字符: \n");     
            scanf("%c", &enter);      //回车键也算字符,所以这里使用其他变量替之.
            scanf("%c", &num);  
            printf("     %c 的 ASCII 为 %d    \n", num, num);
        }
    }
    return 0;
}
