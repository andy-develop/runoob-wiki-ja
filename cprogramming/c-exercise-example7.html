問題：特殊なパターンを出力します。C環境で実行して見てください。Very Beautiful!

プログラム分析：文字は全部で256個あります。文字が異なると、図形も異なります。

VC6.0で中国語の文字化けが発生する（原因＋解決方法）：

176の16進数はB0、219の16進数はDB、0xB0DBは“佰”という文字の内部コードなので、出力されるのは“佰”になる。

主な原因はファイル情報のコードページが異なることです。私たちが使用しているOSの中国語状態のコードページでは、拡張ASCIIコードを表示するには437 OEM-アメリカの下で表示する必要があります。これで希望通りに表示できます。コンソールのデフォルトコードページの具体的な変更手順は次のとおりです。

  • 1. 実行画面の左上隅のタイトルバーアイコン【c:\】をクリックし、[既定値]を選択します。
  • 2. デフォルトのコードページを、936（ANSI/OEM-簡体字中国語GBK）から437 OEM-アメリカに変更します。
  • 3. 閉じてからもう一度実行するだけでOKです。

例

// Created by www.runoob.com on 15/11/9. // Copyright © 2015年 菜鸟教程. All rights reserved. // #include<stdio.h> int main() { char a=176,b=219; printf("%c%c%c%c%c\n",b,a,a,a,b); printf("%c%c%c%c%c\n",a,b,a,b,a); printf("%c%c%c%c%c\n",a,a,b,a,a); printf("%c%c%c%c%c\n",a,b,a,b,a); printf("%c%c%c%c%c\n",b,a,a,a,b); return 0; }

    # Linux環境では、文字'█'（utf-8：9608）、'▒'（utf-8：9618）をルーチンの文字の代わりに使用してみてください。

    #include <stdio.h>

#define LEN 5

int main()
{
    int i, j;
    char *gray_block = "█",      // utf-8字符：9608
         *textured_block = "▒";  // utf-8字符：9618
    for(i=0; i<LEN; i++)  // 行循环
    {
        for(j=0; j<LEN; j++)  // 列循环
            if(i==j || i+j==LEN-1) printf("%s", gray_block);
            else printf("%s", textured_block);
        putchar('\n');  // 换行
    }
    return 0;
}

    エンコーディングを変更するよりも、UTF-8を受け入れる方が良いです。プリプロセッサ機能を使用して、コードがWindowsとLinuxでコンパイル・実行できるようにできます。

    ifdefでマクロが定義されているかどうかを確認して、コンパイルプラットフォームを判断します。

    #include<stdio.h>
#define LEN 5
void print_x(char *highlight_text, char *generic_text);

#ifdef _WIN32 // 判断是不是win32位，64位一般包含这个宏
    #include <Windows.h>
    int main(){
        // 终端编码设置为UTF-8
        SetConsoleOutputCP(65001);
        print_x("█", "▒");
        system("pause");
        return 0;
    }
#elif __linux__ // 判断是不是Linux
    int main(){
        print_x("█", "▒");
        return 0;
    }
#endif

void print_x(char *highlight_text, char *generic_text){
    for (int i = 0; i < LEN; i++){
        for (int j = 0; j < LEN; j++){
            if (i == j || i + j == LEN - 1){
                printf("%s", highlight_text);
            } 
            else {
                printf("%s", generic_text);
            }
        }
        printf("\n");
    }
}
