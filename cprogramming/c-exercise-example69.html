問題：n人の人が輪になって並び、順番に番号を付けます。最初の人から番号を数えます（1から3まで数えます）。3を数えた人は輪から外れます。最後に残るのは元の何番の人かを尋ねます。

プログラム分析：なし。

例

// Created by www.runoob.com on 15/11/9. // Copyright © 2015年 菜鸟教程. All rights reserved. // #include <stdio.h> void main() { int num[50],n,*p,j,loop,i,m,k; printf("この輪の人数を入力してください:\n"); scanf("%d",&n); p=num; //これらの人に番号を付け始める for (j=0;j<n;j++) { *(p+j)=j+1; } i=0;//iはカウントに使う、つまりポインタを後ろに移動させる m=0;//mは輪から外れた人数を記録する k=0;//kは1,2,3と数を数える while(m<n-1)//退出した人数が総人数を超えないとき、つまり残った人数が少なくとも1人であるとき //この文はm<nと書くことはできません。8人と仮定すると、6人が退出したとき、まだ人数が退出する処理を行う、つまりm++です。 //このとき7<8で、残りの1人が自分で1,2,3と数えると、その人も退出してしまい、出力はなくなる { if (*(p+i)!=0)//この人の頭上の番号が0でなければ、カウントを1加える。ここでは、3を数えた人の頭上の番号を0にリセットする方法を採用している { k++; } if (k==3) { k=0; //カウントを0にクリアする、つまり次の人は1から数え始める *(p+i)=0;//3を数えた人の番号を0にリセットする m++; //退出人数を1加える } i++; //ポインタを後ろに移動 if (i==n)//この文は非常に重要で、末尾に達したら、ポインタを先頭に再び指すようにする //そして、それはi++の後に置くしかない。なぜなら、i++して初めてi==nになり得るからだ { i=0; } } printf("今残っている人は:"); for (loop=0;loop<n;loop++) { if (num[loop]!=0) { printf("%2d番\n",num[loop]); } } }

{
    int a[50],n;

    for(int i=0;i<n;i++){
        a[i]=i+1;

        if(a[l]!=0){
            k++;

            a[l]=0;
            coun++;
            k=0;
        }

        l++;
        if(l==n){
            l=0;
        }

    for(int i=0;i<n;i++){
        if(a[i]!=0){
            printf("%d号\n",a[i]);
        }
    }
    return 0;
}

    原理：全員が輪から外れると仮定すると、最後に外れた人が問題で残る一人になる。

    参考方法：

    //vs 2010 pro 编写

#include<stdlib.h>
#include<stdio.h>

void main()
{
    int a[100],n,i,k=0,count=0,lastone;
    printf("请输入人数：");
    scanf("%d",&n);//n为人数
    for(i=0;i<n;i++)//对于每个人a[i],他的编号实际上是i+1
        a[i]=1;/* i表示人的状态，若a[i]=1，则表示i+1号在圈内，
                        若a[i]=0,则表示i+1号在圈外 */
    while(k<n)/* 当所有人都退出圈子时停止循环
              最后退圈的人就是题中有一个留在圈内的那个人 */
    {
        for(i=0;i<n;i++)//对所有编号开始循环
            if(a[i])//判断i+1号是否在圈内
            {
                count++;//若在圈内，则报号
                if(count%3==0)//判断号码是否为3的倍数
                {
                    a[i]=0;//若是3的倍数，则a[i]出圈
                    k++;//退圈人数+1
                                        lastone=i+1;//记录此时退圈人的号码
                }
                                //lastone=i+1 修改这里的位置，放到判断语句内
            }
    }
    printf("留下来的那个是%d号\n",lastone);//将最后退圈的人的号码输出
    system("pause");//防止闪退
}

    動的メモリを使って配列を保存し、ポインタで配列を前方に移動させて要素を削除します。

    #include<stdio.h>
#include<stdlib.h>

void printArr(int *arr, int n){
    int i;
    for(i=0; i<n; i++){
        printf("%d ", *(arr+i));
    }
    printf("\n");
}

// 删除第j个元素，就是第j个元素被后一个元素覆盖
void delArr(int *arr, int n, int j){
    int i;
    for(i=j; i<n; i++){
        *(arr+i)=*(arr+i+1);
    }    
}

int main(){
    int counter=0, n, remain;
    printf("input the number of total people:");
    scanf("%d", &n);
    
    //输入总人数 n
    if(0==n)
        return 1;
    if(n<1){
        printf("Error: must >1\n");
        return 1;
    }
    remain=n;
    //分配一块内存
    int i, *arr0, *arr=malloc( sizeof(int)*n);
    arr0=arr;
    //初始化
    for(i=0; i<n; i++){
        *(arr+i)=(i+1);
    }
    
    int j;
    while(remain>1){
        printf("==> remain=%d (first element counter=%d)\n", remain, counter+1);
        printArr(arr, remain); //剩余元素
        //报数，遇到3的倍数，则
        for(j=0; j<remain; j++){
            counter++;
            counter=counter%3;
            if(counter==0){
                printf("删除元素: %d\n", arr[j]);
                delArr(arr, remain, j); //删除数组arr的第j个元素，后面的前移
                j--;
                remain--;
            }
        }
    }
    printf("剩下的元素是: %d\n", arr[0]);
    return 0;
}
