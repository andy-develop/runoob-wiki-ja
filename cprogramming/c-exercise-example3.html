C言語の定番100例

問題：ある整数に100を加えると完全平方数になり、さらに168を加えると再び完全平方数になります。この数はいくつでしょうか？

プログラム分析：

その数を x と仮定します。

1、すなわち：x + 100 = n2, x + 100 + 168 = m2

2、等式の計算：m2 - n2 = (m + n)(m - n) = 168

3、設定： m + n = i、m - n = j、i * j =168、i と j の少なくとも一方は偶数

4、得られる： m = (i + j) / 2、 n = (i - j) / 2、i と j は両方偶数であるか、両方奇数であるかのいずれか。

5、3 と 4 から導くと、i と j はどちらも 2 以上の偶数であることがわかる。

6、i * j = 168、 j>=2 であるため、1 < i < 168 / 2 + 1。

7、次に i のすべての数値をループで計算すればよい。

具体的な実装は次のとおりです：

例

#include <stdio.h> int main (void) { int i, j, m, n, x; for (i = 1; i < 168 / 2 + 1; i++) { if (168 % i == 0) { j = 168 / i; if ( i > j && (i + j) % 2 == 0 && (i - j) % 2 == 0) { m = (i + j) / 2; n = (i - j) / 2; x = n * n - 100; printf ("%d + 100 = %d * %d\n", x, n, n); printf ("%d + 268 = %d * %d\n", x, m, m); } } } return 0; }

上記の例の出力結果は次のとおりです：

-99 + 100 = 1 * 1
-99 + 268 = 13 * 13
21 + 100 = 11 * 11
21 + 268 = 17 * 17
261 + 100 = 19 * 19
261 + 268 = 23 * 23
1581 + 100 = 41 * 41
1581 + 268 = 43 * 43

    考え方：

    m^2 - n^2 = (m + n)(m - n) = 168 から、|m|>|n| が得られる。

    上式で |n| が最大のとき、(n+1)^2 - n^2 ≥ 168 → n ≥ 83.5 となります。したがって、n の値の範囲を 1〜84 と選びます。

    このとき、x=n^2-100、m=(x+268)^0.5 となります。各 n に対応する x と m を求め、その中から整数型の x と m を選んで出力すればよい。

    プログラム：

    #include <stdio.h>
#include <math.h> 

int main()
{
    int n,x,a,m;
    double b;

    for (n = 1; n <= 168 / 2 ; n++)
    {
        x=n*n-100;
        a=x+268;
        b=sqrt(a);
        if((int)b==b)
        {
            m=(int)b;
            printf ("%d + 100 = %d * %d\n", x, n, n);
            printf ("%d + 268 = %d * %d\n", x, m, m);
        }
    }
    return 0;
}

    /*
  思路：
    1.根据已知条件判断该整数 x 的上下限
        由已知条件得
            sqrt(x + 100) = y    由 y>=0 可知 x 的下限即为 -100
            sqrt(x + 268) = z
        由上面两式可得
            z^2 - y^2 = 168
        按照两个相邻整数的平方差间隔越来越大的特性，找出极限情况下的数值 y 与 z
        将 z^2 - 268 或 y^2 - 100 作为 x 的上限
    2.然后在范围内暴力搜索，找出所有符合描述的值
        根据 x 的值计算 y, z
        判断 y, z 均为整数，即输出 x
*/

# include <stdio.h>
# include <math.h>

int main()
{
    int x, c1, c2, L, H;
    c1 = 100;
    c2 = 168;

    L = -c1;

    int i = 0;    
    for (i = 1; i*i - (i - 1)*(i - 1) <= c2; ++i)
        ;

    H = i*i - (c1 + c2);

    int j;
    double y, z;
    for (j = L; j <= H; ++j)
    {
        y = sqrt(j + c1);
        z = sqrt(j + c1 + c2);
        if (y == (int)y && z == (int)z)
            printf("x的值为 %d\n", j);
    }

    return 0;
}

    私の考え方はこうです：鍵となるのは2つの平方数の差が168であることです。そこで条件を満たすすべての平方数を探索します。i を小さい方、j を大きい方とします。j*j - i*i = 168 を満たすとき、x は i*i - 100 に等しくなります。

    i を探索するとき、(i+1) の平方から i の平方を引いた値が 168 より大きければ、探索する必要はありません。条件を満たすことはあり得ないからです。j を探索するときは、i+1 から探索します。j の平方 - i の平方 が 168 より大きければ、それも探索する必要はありません。

    #include<stdio.h>
#include<math.h>

int main(void){
    for (int i = 0; (i+1)*(i+1)-i*i <= 168 ; ++i)
    {
        for (int j = i+1; j*j-i*i<=168 ; ++j)
        {
            if(j*j-i*i==168) {
                printf("j = %d,i = %d,j*j-i*i=%d\n", j,i,j*j-i*i);
                printf("x = %d\n", i*i-100);
            }
            
        }
    }
}

    #include <stdio.h>
#include <math.h>

int main(){
    int y,i;//记i=x-y,(x+y)(x-y)=168
    for(i=2;i<=sqrt(168);i=i+2){  //在2~sqrt(168)的偶数之间找(x-y)的可能值。
        if(84%i==0){  //这里不用168%i==0是因为由于(x+y)也是偶数，缩小一下范围。
            y=(168/i-i)/2;  //168/i=(x+y),i=(x-y)
            printf("这个数可以是 %d\n",y*y-100);
        }
    }
}
