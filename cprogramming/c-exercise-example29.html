問題：5桁以下の正の整数を1つ与えます。要件：一、それが何桁かを求める。二、各桁の数字を逆順に出力する。

プログラム分析：各桁の数字に分解することを学びます。以下の説明の通りです。

例

// Created by www.runoob.com on 15/11/9. // Copyright © 2015年 菜鸟教程. All rights reserved. // #include <stdio.h> int main( ) { long a,b,c,d,e,x; printf("5桁の数字を入力してください："); scanf("%ld",&x); a=x/10000; /*万の位を分解*/ b=x%10000/1000; /*千の位を分解*/ c=x%1000/100; /*百の位を分解*/ d=x%100/10; /*十の位を分解*/ e=x%10; /*一の位を分解*/ if (a!=0){ printf("5桁の数です。逆順： %ld %ld %ld %ld %ld\n",e,d,c,b,a); } else if(b!=0) { printf("4桁の数です。逆順： %ld %ld %ld %ld\n",e,d,c,b); } else if(c!=0) { printf("3桁の数です。逆順：%ld %ld %ld\n",e,d,c); } else if(d!=0) { printf("2桁の数です。逆順： %ld %ld\n",e,d); } else if(e!=0) { printf("1桁の数です。逆順：%ld\n",e); } }

上記の例の出力結果は次のとおりです：

请输入 5 位数字：12345
为 5 位数,逆序为： 5 4 3 2 1

int reverse(int num, char* buf)
{
    int head = 0, tail = num - 1;
    int counter = num / 2;
    if(counter == 0)
        return 0;
    *(buf + head) ^= *(buf + tail);
    *(buf + tail) ^= *(buf + head);
    *(buf + head) ^= *(buf + tail);
    return reverse(num - 2, buf + 1);
}
int main()
{
    char buf[6] = {0};

    printf("逆序输出：");
    reverseprintNum(num);
    return 0;
}

int digits(int k)
{
    if (k / 10 == 0) return 1;
    else return digits(k / 10) + 1;
}

void reverseprintNum(int g)
    {
    if (g / 10 == 0) printf("%d", g);
    else
    {
        printf("%d", g % 10);     // 最后一个数字作为第一个数字输出
        reverseprintNum(g / 10);  // 删除最后一个数字
    }
}

    示されている例には少し問題があります。5桁を超える数字を入力した場合、逆順に出力はされますが、最後の数字が複数桁になることがあります。例えば以下のような場合です：

    请输入 5 位数字：000123456789
为 5 位数,逆序为： 9 8 7 6 12345

    0 も整数ですが、その判定に従うと、0 はどの条件も満たさないため、特殊な場合の処理がありません。

    以下の方法を参考にできます：

    /*
  1、输入以字符串的形式获取
  2、获取字符串之后，自定义一个函数，把开头的0去掉
  3、根据自定义函数返回值的长度进行处理
    （1）返回值长度为0，给字符串就赋值为0
    （2）长度大于0并且小于5，不做额外处理
    （3）长度大于5，只保留前五位，并且认为长度就为5
  4、利用for循环进行逆序输出
  5、如果想要处理更多位数的数字，只需要修改宏定义即可
*/
#include <stdio.h>
#include <string.h>

#define NUM_LENGTH 5            //超过5位数保留前5位

char *deal_zero(char *str, int len);            //把输入的字符串前边的所有0去掉

int main( )
{
    unsigned char str[32] ={0} ;
    unsigned char *temp_str = NULL;
    unsigned char result_str[32] = {0};
    int i = 0, len = 0;
    printf("请输入 %d 位数字：", NUM_LENGTH);
        scanf("%s", str);
    temp_str = deal_zero(str, strlen(str));
    len = strlen(temp_str);
    if(len == 0)
    {
        len = 1;
        result_str[0] = '0';
    }
    else if(len > NUM_LENGTH)
    {
        len = NUM_LENGTH;
        strncpy(result_str, temp_str, len);
    }
    else
        strncpy(result_str, temp_str, len);
    printf("这是一个%d位数，逆序输出 : ", len);
    for(i = 0; i < len; i++)
    {
        printf("%c ", result_str[len - i - 1]);
    }
    printf("\n");
    return 0;
}

char *deal_zero(char *str, int len)
{
    if(str[0] == '0')
    {
        strncpy(str, str+1, len--);
        deal_zero(str, len--);
    }
    return str;
}

/*
给一个不多于5位的正整数，要求：一、求它是几位数，二、逆序打印出各位数字
*/
int count = 0;
void p(){
    char ch;
    if ((ch = getchar()) != '\n') {
        count = count + 1;
        p();
    }
    if (ch != '\n') {
        printf("%c",ch);
    }
}

int main() {
    p();
    printf("%d",count);
    while (1);
    return 0;
}

    X を変更すると、数値の範囲を 10^X まで拡大でき、さらに桁数を示すこともできます。

    #include <stdio.h>
#include <math.h>
#define X 5 
int main()
{
    int num,bit[X],i,j;
    scanf("%d",&num);
    if(num<0||num>int(pow(10,X))) //出错判断 
        printf("error");
    else
    {    
        for(i=0;num!=0;i++)
        {
            bit[i]=num%10;//存储每一位 
            num=(num-num%10)/10;
        }
        printf("这是一个%d位数\n",i);
        for(j=0;j<=i-1;j++)
            printf("10^%d位：%d\n",j,bit[j]);
    }        
    return 0;
}
