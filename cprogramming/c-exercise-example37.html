C 練習例37 - ソート

C言語の定番100例

問題：10個の数値をソートします。

プログラム解析：選択法が利用できます。すなわち、後ろの9個の要素との比較で最小のものを選び、最初の要素と交換します。次回も同様に、2番目の要素と後ろの8個を比較して交換します。

例

#include <stdio.h> #define N 10 void selectionSort(int arr[], int n) { int i, j, minIndex, temp; for (i = 0; i < n - 1; i++) { minIndex = i; for (j = i + 1; j < n; j++) { if (arr[j] < arr[minIndex]) { minIndex = j; } } if (minIndex != i) { // Swap the elements temp = arr[i]; arr[i] = arr[minIndex]; arr[minIndex] = temp; } } } int main() { int arr[N]; printf("%d 個の数値を入力してください：\n", N); for (int i = 0; i < N; i++) { scanf("%d", &arr[i]); } selectionSort(arr, N); printf("ソート結果は：\n"); for (int i = 0; i < N; i++) { printf("%d ", arr[i]); } printf("\n"); return 0; }

上記の例の出力結果は：

请输入 10 个数字：
23 2 27 98 234 1 4 90 88 34
排序结果是:
1 2 4 23 27 34 88 90 98 234

    基礎アルゴリズムの単純選択ソート（selection sort）

    1，名 称：単純選択ソート

    2、計算量：O(n^2)

    3、実装方式：C言語

    4、空間計算量：O(1)

    5、安定性：不安定

    6，アルゴリズムの考え方：全体で2回ループします。外側のループはアルゴリズムが合計で実行する回数です。では、なぜでしょうか？このアルゴリズムは外側のループを1回実行するごとに1回交換を行い、既定ではiの位置が最大値または最小値（昇順か降順かによって決まる）とみなします。そして内側のループが交換する数値を決定します。具体的な考え方は、コードを読んで実感してください！

    7，アルゴリズムの種類：昇順（ascending order）、降順（descending order）

void swap(int *a, int *b){  
    int temp ;  
    temp = *a;  
    *a = *b;  
    *b = temp;  
}  
//升序---简单选择排序  
SelectionSortAsc(int *a,int n){  
    int i,j,min;  
    for( i = 0;i < 10; i++){  
        min = i;  
        for(j = i+1;j < 10; j++){  
            if(a[j] < a[min]){  
                min = j;      
            }     
        }  
        if(min != i)          
        swap(&a[i],&a[min]);  
    }  
    printf("\n  ascending order：");                   
}   
//降序---简单选择排序  
SelectionSortDesc(int *a,int n){  
    int i,j,max;  
    for( i = 0;i < 10; i++){  
        max = i;  
        for(j = i+1;j < 10; j++){  
            if(a[j] > a[max]){  
                max = j;      
            }     
        }  
        if(max != i)          
        swap(&a[i],&a[max]);  
    }  
    printf("\n  descending order：");                  
}   
 main(){

void bubblesort(int a[],int n)
{
    int i, j;
    int flag ;
    for(i = 0; i < n - 1; i++)
    {
        flag = 0;
        for(j = 0; j < n - 1; j++)
        {
            int temp;
            if(a[j] > a[j + 1])
            {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
                flag = 1;
            }
        }
        if(flag == 0)
            break;
    }
}

    #include <stdio.h>

typedef void (*p_func)(int *, int);

void select_sort(int  *arr, int size)   //选择排序
{
    int i = 0, j = 0;
    for(i = 0; i < size; i++)
    {
        for(j = i; j < size; j++)
        {
            if(arr[i] > arr[j] )
            {
                arr[i] ^= arr[j];
                arr[j] ^= arr[i];
                arr[i] ^= arr[j];
            }
        }
    }
}

void bubble_sort(int *arr, int size)    //冒泡排序
{
    int i = 0, j = 0;
    for(i = 1; i < size; i++)
    {
        for(j = 0; j < size - i; j++)
        {
            if(arr[j] > arr[j+1])
            {
                arr[j] ^= arr[j+1];
                arr[j+1] ^= arr[j];
                arr[j] ^= arr[j+1];
            }   
        }   
    } 
}
void quick_sort(int *arr, int size)     //快速排序
{
    if(size <= 1)
        return;
    int base = *arr;
    int head = 0, tail = size - 1;
    while(head < tail)
    {
        if(arr[head] > arr[tail])
        {
            arr[head] ^= arr[tail];
            arr[tail] ^= arr[head];
            arr[head] ^= arr[tail];
        }
        if(arr[head] == base)
            tail--;
        else head++;
    }
    quick_sort(arr, head - 1);
    quick_sort(&arr[head + 1], size - head - 1);
}
int main()
{
    int arr[] = {0, 4, 2, 8, 6, 1, 5, 9, 3, 7};
    int i = 0, j = 0;
    p_func p_sort[] = {select_sort, bubble_sort, quick_sort};
    for(i = 0; i <= 2; i++)
    {
        p_sort[i](arr, sizeof(arr)/sizeof(int));
        i == 0 ? printf("select : ") : i == 1 ? printf("bubble : ") : printf("quick  : ");
        for(j = 0; j < sizeof(arr)/sizeof(int); j++)
            printf("%d  ", arr[j]);
        printf("\n");
    }

    return 0;
}
