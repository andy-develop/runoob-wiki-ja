C# 运算符重载

C# の組み込み演算子は再定義またはオーバーロードできます。したがって、プログラマーはユーザー定義型に対しても演算子を使用できます。オーバーロードされた演算子は、特別な名前を持つ関数であり、キーワードoperatorの後に演算子の記号を続けて定義されます。他の関数と同様に、オーバーロードされた演算子には戻り値の型とパラメーターリストがあります。

例えば、次の関数を見てください：

public static Box operator+ (Box b, Box c)
{
   Box box = new Box();
   box.length = b.length + c.length;
   box.breadth = b.breadth + c.breadth;
   box.height = b.height + c.height;
   return box;
}

上記の関数は、ユーザー定義のクラス Box に加算演算子（+）を実装します。これは2つの Box オブジェクトのプロパティを加算し、加算後の Box オブジェクトを返します。

演算子オーバーロードの実装

以下のプログラムは完全な実装を示しています：

实例

using System;

namespace OperatorOvlApplication
{
   class Box
   {
      private double length;      // 长度
      private double breadth;     // 宽度
      private double height;      // 高度

      public double getVolume()
      {
         return length * breadth * height;
      }
      public void setLength( double len )
      {
         length = len;
      }

      public void setBreadth( double bre )
      {
         breadth = bre;
      }

      public void setHeight( double hei )
      {
         height = hei;
      }
      // 重载 + 运算符来把两个 Box 对象相加
      public static Box operator+ (Box b, Box c)
      {
         Box box = new Box();
         box.length = b.length + c.length;
         box.breadth = b.breadth + c.breadth;
         box.height = b.height + c.height;
         return box;
      }

   }

   class Tester
   {
      static void Main(string[] args)
      {
         Box Box1 = new Box();         // 声明 Box1，类型为 Box
         Box Box2 = new Box();         // 声明 Box2，类型为 Box
         Box Box3 = new Box();         // 声明 Box3，类型为 Box
         double volume = 0.0;          // 体积

         // Box1 详述
         Box1.setLength(6.0);
         Box1.setBreadth(7.0);
         Box1.setHeight(5.0);

         // Box2 详述
         Box2.setLength(12.0);
         Box2.setBreadth(13.0);
         Box2.setHeight(10.0);

         // Box1 的体积
         volume = Box1.getVolume();
         Console.WriteLine("Box1 的体积： {0}", volume);

         // Box2 的体积
         volume = Box2.getVolume();
         Console.WriteLine("Box2 的体积： {0}", volume);

         // 把两个对象相加
         Box3 = Box1 + Box2;

         // Box3 的体积
         volume = Box3.getVolume();
         Console.WriteLine("Box3 的体积： {0}", volume);
         Console.ReadKey();
      }
   }
}

上記のコードがコンパイルされ実行されると、次の結果が生成されます：

Box1 的体积： 210
Box2 的体积： 1560
Box3 的体积： 5400

オーバーロード可能および不可能な演算子

次の表は、C# における演算子オーバーロードの機能を示しています。

运算符描述
+, -, !, ~, ++, --これらの単項演算子はオペランドを1つだけ持ち、オーバーロード可能です。
+, -, *, /, %これらの二項演算子は2つのオペランドを持ち、オーバーロード可能です。
==, !=, <, >, <=, >=これらの比較演算子はオーバーロードできます。
&&, ||これらの条件付き論理演算子は直接オーバーロードできません。
+=, -=, *=, /=, %=これらの代入演算子はオーバーロードできません。
=, ., ?:, ->, new, is, sizeof, typeofこれらの演算子はオーバーロードできません。

实例

上記の議論に基づいて、上記の例を拡張して、さらに多くの演算子をオーバーロードしてみましょう：

实例

using System;

namespace OperatorOvlApplication
{
    class Box
    {
       private double length;      // 长度
       private double breadth;     // 宽度
       private double height;      // 高度
     
       public double getVolume()
       {
         return length * breadth * height;
       }
      public void setLength( double len )
      {
          length = len;
      }

      public void setBreadth( double bre )
      {
          breadth = bre;
      }

      public void setHeight( double hei )
      {
          height = hei;
      }
      // 重载 + 运算符来把两个 Box 对象相加
      public static Box operator+ (Box b, Box c)
      {
          Box box = new Box();
          box.length = b.length + c.length;
          box.breadth = b.breadth + c.breadth;
          box.height = b.height + c.height;
          return box;
      }
     
      public static bool operator == (Box lhs, Box rhs)
      {
          bool status = false;
          if (lhs.length == rhs.length && lhs.height == rhs.height
             && lhs.breadth == rhs.breadth)
          {
              status = true;
          }
          return status;
      }
      public static bool operator !=(Box lhs, Box rhs)
      {
          bool status = false;
          if (lhs.length != rhs.length || lhs.height != rhs.height
              || lhs.breadth != rhs.breadth)
          {
              status = true;
          }
          return status;
      }
      public static bool operator <(Box lhs, Box rhs)
      {
          bool status = false;
          if (lhs.length < rhs.length && lhs.height
              < rhs.height && lhs.breadth < rhs.breadth)
          {
              status = true;
          }
          return status;
      }

      public static bool operator >(Box lhs, Box rhs)
      {
          bool status = false;
          if (lhs.length > rhs.length && lhs.height
              > rhs.height && lhs.breadth > rhs.breadth)
          {
              status = true;
          }
          return status;
      }

      public static bool operator <=(Box lhs, Box rhs)
      {
          bool status = false;
          if (lhs.length <= rhs.length && lhs.height
              <= rhs.height && lhs.breadth <= rhs.breadth)
          {
              status = true;
          }
          return status;
      }

      public static bool operator >=(Box lhs, Box rhs)
      {
          bool status = false;
          if (lhs.length >= rhs.length && lhs.height
             >= rhs.height && lhs.breadth >= rhs.breadth)
          {
              status = true;
          }
          return status;
      }
      public override string ToString()
      {
          return String.Format("({0}, {1}, {2})", length, breadth, height);
      }
   
   }
   
   class Tester
   {
      static void Main(string[] args)
      {
        Box Box1 = new Box();          // 声明 Box1，类型为 Box
        Box Box2 = new Box();          // 声明 Box2，类型为 Box
        Box Box3 = new Box();          // 声明 Box3，类型为 Box
        Box Box4 = new Box();
        double volume = 0.0;   // 体积

        // Box1 详述
        Box1.setLength(6.0);
        Box1.setBreadth(7.0);
        Box1.setHeight(5.0);

        // Box2 详述
        Box2.setLength(12.0);
        Box2.setBreadth(13.0);
        Box2.setHeight(10.0);

       // 使用重载的 ToString() 显示两个盒子
        Console.WriteLine("Box1： {0}", Box1.ToString());
        Console.WriteLine("Box2： {0}", Box2.ToString());
       
        // Box1 的体积
        volume = Box1.getVolume();
        Console.WriteLine("Box1 的体积： {0}", volume);

        // Box2 的体积
        volume = Box2.getVolume();
        Console.WriteLine("Box2 的体积： {0}", volume);

        // 把两个对象相加
        Box3 = Box1 + Box2;
        Console.WriteLine("Box3： {0}", Box3.ToString());
        // Box3 的体积
        volume = Box3.getVolume();
        Console.WriteLine("Box3 的体积： {0}", volume);

        //comparing the boxes
        if (Box1 > Box2)
          Console.WriteLine("Box1 大于 Box2");
        else
          Console.WriteLine("Box1 不大于 Box2");
        if (Box1 < Box2)
          Console.WriteLine("Box1 小于 Box2");
        else
          Console.WriteLine("Box1 不小于 Box2");
        if (Box1 >= Box2)
          Console.WriteLine("Box1 大于等于 Box2");
        else
          Console.WriteLine("Box1 不大于等于 Box2");
        if (Box1 <= Box2)
          Console.WriteLine("Box1 小于等于 Box2");
        else
          Console.WriteLine("Box1 不小于等于 Box2");
        if (Box1 != Box2)
          Console.WriteLine("Box1 不等于 Box2");
        else
          Console.WriteLine("Box1 等于 Box2");
        Box4 = Box3;
        if (Box3 == Box4)
          Console.WriteLine("Box3 等于 Box4");
        else
          Console.WriteLine("Box3 不等于 Box4");

        Console.ReadKey();
      }
    }
}

上記のコードがコンパイルされ実行されると、次の結果が生成されます：

Box1： (6, 7, 5)
Box2： (12, 13, 10)
Box1 的体积： 210
Box2 的体积： 1560
Box3： (18, 20, 15)
Box3 的体积： 5400
Box1 不大于 Box2
Box1 小于 Box2
Box1 不大于等于 Box2
Box1 小于等于 Box2
Box1 不等于 Box2
Box3 等于 Box4

    operator キーワードは、クラスまたは構造体の宣言内で演算子を宣言するために使用されます。演算子の宣言は、次の4つの形式のいずれかを使用できます。

    public static result-type operator unary-operator ( op-type operand )
public static result-type operator binary-operator ( op-type operand, op-type2 operand2 )
public static implicit operator conv-type-out ( conv-type-in operand )
public static explicit operator conv-type-out ( conv-type-in operand )

    参数：

    • result-type は演算子の結果型です。
    • unary-operator 下列运算符之一：+ - ! ~ ++ — true false
    • op-type 最初の（または唯一の）パラメーターの型。
    • operand 最初の（または唯一の）パラメーターの名前。
    • binary-operator 其中一个：+ - * / % & | ^ << >> == != > < >= <=
    • op-type2 2番目のパラメーターの型。
    • operand2 2番目のパラメーターの名前。
    • conv-type-out 型変換演算子のターゲット型。
    • conv-type-in 型変換演算子の入力型。

    注意：

    最初の2つの形式は、組み込み演算子をオーバーロードするユーザー定義の演算子を宣言します。すべての組み込み演算子をオーバーロードできるわけではありません（「オーバーロード可能な演算子」を参照してください）。op-type と op-type2 の少なくとも1つは、包含型（つまり、演算子が属する型、またはユーザー定義型と見なされる型）である必要があります。これにより、たとえば整数の加算演算子を再定義することはできません。

    後の2つの形式は、変換演算子を宣言します。conv-type-in と conv-type-out のうち、ちょうど1つは包含型である必要があります（つまり、変換演算子は、その包含型から他の型への変換、または他の型からその包含型への変換のみを行うことができます）。

    演算子は値パラメーターのみを受け取ることができ、ref または out パラメーターを受け取ることはできません。

    C# では、比較演算子をペアでオーバーロードする必要があります。== をオーバーロードした場合は、!= もオーバーロードする必要があります。そうしないとコンパイルエラーが発生します。また、比較演算子は bool 型の値を返す必要があります。これは他の算術演算子との根本的な違いです。

    C# では = 演算子をオーバーロードすることはできませんが、たとえば + 演算子をオーバーロードした場合、コンパイラーは自動的に + 演算子のオーバーロードを使用して += 演算子の操作を実行します。

    演算子のオーバーロードは、実際には関数のオーバーロードです。まず、指定された演算式によって対応する演算子関数が呼び出され、次にオペランドが演算子関数の実引数に変換され、続いて実引数の型に基づいて呼び出す関数のオーバーロードが決定されます。このプロセスはコンパイラーによって実行されます。

    任意の演算子宣言の前に、省略可能な属性（C# プログラミングガイド）のリストを付けることができます。

    using System;
using System.Collections.Generic;
using System.Text;

namespace OperatorOverLoading
{
    class Program
    {
        static void Main(string[] args)
        {
            Student s1 = new Student(20, "Tom");
            Student s2 = new Student(18, "Jack");
            Student s3 = s1 + s2;

            s3.sayPlus();
            (s1 - s2).sayMinus();
            Console.ReadKey();
        }
    }
    public class Student
    {
        public Student() { }
        public Student(int age, string name)
        {
            this.name = name;
            this.age = age;

        }
        private string name;
        private int age;

        public void sayPlus()
        {
            System.Console.WriteLine("{0} 年龄之和为：{1}", this.name, this.age);

        }
        public void sayMinus() {
            System.Console.WriteLine("{0} 年龄之差为：{1}", this.name, this.age);
        }
        //覆盖“+”操作符
        public static Student operator +(Student s1, Student s2)
        {
            return new Student(s1.age + s2.age, s1.name + " And " + s2.name);

        }
        //覆盖“-”操作符
        public static Student operator -(Student s1, Student s2) {
            return new Student(Math.Abs(s1.age - s2.age), s1.name + "And" + s2.name);
        }
   }
}
