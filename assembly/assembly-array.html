アセンブリ言語 - 配列

配列は、同じ型のデータを連続して格納した集合です。アセンブリ言語では、配列はメモリ上の連続した領域であり、ベースアドレス＋オフセットの方式で各要素にアクセスします。

一次元配列の定義

实例

; 配列の定義方法

section .data
    ; 方式1：逐个列出元素
    arr1 dd 10, 20, 30, 40, 50  ; 5 つのダブルワード要素の配列

    ; 方式2：用 dup 初始化
    arr2 dd 10 dup(0)            ; 10 個のダブルワード、すべて 0

    ; 方式3：字节数组
    bytes db 1, 2, 3, 4, 5, 6   ; 6 个字节

    ; 方式4：字符数组（字符串）
    chars db 'runoob', 0         ; 7 字节

    ; 数组长度计算（编译时）
    arr1_len equ ($ - arr1) / 4  ; ダブルワード配列の要素数
    arr2_len equ ($ - arr2) / 4
    bytes_len equ ($ - bytes)    ; 字节数组，不需除法

section .bss
    ; 未初期化の配列
    buffer resd 100              ; 100 個のダブルワードの領域を予約

配列要素のアクセス

配列要素のアクセスに使用する基址 + 索引 × 元素大小アドレス指定方式：

实例

; 文件路径：array_access.asm
; 配列要素の読み取りと書き込み

section .data
    nums dd 100, 200, 300, 400, 500   ; 5 个元素
    nums_len equ ($ - nums) / 4

section .text
    global _start

_start:
    ; 访问第 0 个元素（下标 0）
    mov eax, [nums]              ; eax = 100

    ; 访问第 2 个元素（下标 2）
    mov eax, [nums + 2 * 4]      ; eax = nums[2] = 300
    ; 2 * 4 = 8、nums から 8 バイトオフセット

    ; レジスタをインデックスとして使用する
    mov esi, 3                   ; 下标 = 3
    mov eax, [nums + esi * 4]    ; eax = nums[3] = 400

    ; 修改数组元素
    mov dword [nums + 4], 250    ; nums[1] = 250
    ; 配列の中身は現在：100, 250, 300, 400, 500

    ; EBX をベースレジスタとして使用
    mov ebx, nums                ; ebx = 数组基址
    mov eax, [ebx + 4 * 4]       ; eax = nums[4] = 500

    mov eax, 1
    mov ebx, 0
    int 0x80

注意：[nums + esi * 4]中で4を掛けるのは、各要素が4バイト（ダブルワード）であるためです。バイト配列（db）の場合は、直接次のように書きます。[arr + esi]；ワード配列（dw）、次のように書く。[arr + esi * 2]。

数组遍历

实例

; 文件路径：array_traverse.asm
; 配列を走査して合計を計算する

section .data
    numbers dd 5, 10, 15, 20, 25, 30, 35, 40, 45, 50
    count equ ($ - numbers) / 4   ; 元素个数

section .text
    global _start

_start:
    mov ecx, count                ; 循环次数
    mov esi, 0                    ; インデックス（0 から始まる）
    mov eax, 0                    ; 累積と

sum_loop:
    add eax, [numbers + esi * 4]  ; 累加数组元素
    inc esi                       ; 下标 +1
    loop sum_loop
    ; eax = 5+10+15+...+50 = 275

    ; 另一种遍历方式：使用指针
    mov ecx, count
    mov ebx, numbers              ; ebx 指向数组起始
    mov eax, 0                    ; 累積と

sum_loop2:
    add eax, [ebx]                ; 累加当前元素
    add ebx, 4                    ; 指针移动到下一个元素
    loop sum_loop2
    ; eax = 275

    mov ebx, eax                  ; 終了コード = 累積と
    mov eax, 1
    int 0x80

数组查找

实例

; 文件路径：array_search.asm
; 配列内で指定値を検索する

section .data
    data dd 12, 45, 67, 23, 89, 34, 56, 78, 90, 11
    data_len equ ($ - data) / 4
    target dd 23                   ; 検索する値
    found_msg db 'Found at index: ', 0
    found_len equ $ - found_msg
    notfound_msg db 'Not found (runoob)', 0xA
    notfound_len equ $ - notfound_msg
    newline db 0xA

section .text
    global _start

_start:
    mov ecx, data_len             ; 循环次数
    mov esi, 0                    ; 当前下标

search_loop:
    mov eax, [data + esi * 4]     ; 加载当前元素
    cmp eax, [target]             ; 目標値に等しいかどうか
    je  found                     ; 見つかった

    inc esi                       ; 下标 +1
    loop search_loop

    ; 没找到
    mov eax, 4
    mov ebx, 1
    mov ecx, notfound_msg
    mov edx, notfound_len
    int 0x80
    jmp exit

found:
    ; 見つかった（esi は添字）
    mov eax, 4
    mov ebx, 1
    mov ecx, found_msg
    mov edx, found_len
    int 0x80

    ; 添字をASCII文字に変換して出力する
    add esi, '0'                  ; 单数字下标转字符
    push esi                      ; 一時保存としてスタックにプッシュ
    mov eax, 4
    mov ebx, 1
    mov ecx, esp                  ; 栈顶地址
    mov edx, 1
    int 0x80
    pop esi

    ; 输出换行
    mov eax, 4
    mov ebx, 1
    mov ecx, newline
    mov edx, 1
    int 0x80

exit:
    mov eax, 1
    mov ebx, 0
    int 0x80

二维数组

2次元配列はメモリ上で行ごとに1次元として展開されて格納されます。アクセスは...arr[i][j]のアドレス計算式は：

地址 = 基址 + (i*列数 + j) * 元素大小

实例

; 文件路径：2d_array.asm
; 2次元配列の定義とアクセス

section .data
    ; 3行4列の二次元配列
    matrix dd 1,  2,  3,  4
           dd 5,  6,  7,  8
           dd 9, 10, 11, 12

    rows equ 3
    cols equ 4
    elem_size equ 4              ; 双字 = 4 字节

section .text
    global _start

_start:
    ; 访问 matrix[1][2] = 7（第2行第3列）
    ; 地址 = matrix + (1*4 + 2) * 4 = matrix + 24

    mov eax, [matrix + (1 * cols + 2) * elem_size]
    ; eax = 7

    ; 使用寄存器动态计算（假设 i=2, j=1）
    ; matrix[2][1] = 10（第3行第2列）
    mov esi, 2                   ; 行 i = 2
    mov edi, 1                   ; 列 j = 1

    mov eax, cols                ; 列数
    mul esi                      ; eax = i * cols = 2*4 = 8
    add eax, edi                 ; eax = i*cols + j = 8+1 = 9
    ; eax = eax * elem_size
    ; 注意：MUL の結果は EAX にあり、ここでは直接使用する
    mov eax, [matrix + eax * elem_size]
    ; eax = 10

    ; 遍历二维数组所有元素
    mov ecx, rows * cols         ; 总元素数 = 12
    mov esi, 0                   ; 下标
    mov ebx, 0                   ; 累積と

traverse:
    add ebx, [matrix + esi * elem_size]
    inc esi
    loop traverse
    ; ebx = 1+2+3+...+12 = 78

    mov eax, 1
    int 0x80

バブルソートの完全な例

实例

; 文件路径：bubble_sort.asm
; 冒泡排序算法

section .data
    array dd 64, 34, 25, 12, 22, 11, 90, 78
    array_len equ ($ - array) / 4

section .text
    global _start

_start:
    mov ecx, array_len            ; 外层循环：n 次
    dec ecx                       ; 外层只需 n-1 次

outer_loop:
    push ecx                      ; 保存外层计数器

    mov esi, 0                    ; 内側のインデックスは 0 から始まる
    mov ecx, array_len - 1        ; 内层循环次数

inner_loop:
    mov eax, [array + esi * 4]    ; a[j]
    mov ebx, [array + esi * 4 + 4]  ; a[j+1]
    cmp eax, ebx                  ; a[j] > a[j+1] ?
    jle no_swap                   ; 否，不交换

    ; a[j] と a[j+1] を交換する
    mov [array + esi * 4], ebx
    mov [array + esi * 4 + 4], eax

no_swap:
    inc esi
    loop inner_loop

    pop ecx
    loop outer_loop
    ; 配列は現在ソート済み: 11, 12, 22, 25, 34, 64, 78, 90

    mov eax, 1
    mov ebx, 0
    int 0x80

アセンブリ言語の配列には境界チェックが一切ありません。範囲外のインデックスにアクセスしてもエラーにはならず、隣接するメモリ上のデータを静かに読み書きしてしまいます。これはデバッグが困難なバグの原因となり得ます。必ず自分でインデックスが正当な範囲内にあることを確認してください。
