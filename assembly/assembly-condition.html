アセンブリ言語 - 条件分岐

条件判断はプログラム制御フローの基盤である。アセンブリ言語では比較命令と条件分岐命令を用いて、if-else、switch などの判断ロジックを実現する。

CMP - 比較命令

CMP2つのオペランドで減算（デスティネーション - ソース）を行うが、結果は保存せず、フラグのみ更新する。

本质上CMP dest, src等同于SUB dest, srcしかし、計算結果は破棄される。

实例

; CMP 比较指令示例

section .text
    global _start

_start:
    mov eax, 10
    cmp eax, 10                 ; eax == 10?
    ; ZF = 1（相等，结果为零）
    ; CF = 0（无借位）
    ; SF = 0（结果非负）

    mov eax, 5
    cmp eax, 10                 ; eax == 10?
    ; ZF = 0（不等，结果非零）
    ; CF = 1（有借位：5 < 10）

    mov eax, 20
    cmp eax, 10                 ; eax == 10?
    ; ZF = 0（不等）
    ; CF = 0（无借位：20 >= 10）
    ; SF = 0（结果非负：10 > 0）

    mov eax, 1
    mov ebx, 0
    int 0x80

条件分岐命令

条件分岐命令はフラグレジスタの状態に基づいて分岐するかどうかを決定する。条件が成立すればターゲットラベルにジャンプし、成立しなければ次の命令を実行し続ける。

指令含义チェックされるフラグ适用场景
JE / JZ等しい / ゼロであるときにジャンプZF = 1cmp a, b後で a == b をチェック
JNE / JNZ等しくない / ゼロでないときにジャンプZF = 0cmp a, b後で a != b をチェック
JG / JNLEより大きいときにジャンプ（符号あり）ZF=0 且 SF=OF符号付き数値 a > b
JGE / JNL以上であるときにジャンプ（符号あり）SF = OF符号付き数値 a >= b
JL / JNGEより小さいときにジャンプ（符号あり）SF != OF符号付き数値 a < b
JLE / JNG以下であるときにジャンプ（符号あり）ZF=1 或 SF!=OF符号付き数値 a <= b
JA / JNBEより大きいときにジャンプ（符号なし）CF=0 且 ZF=0符号なし数値 a > b
JAE / JNB以上であるときにジャンプ（符号なし）CF = 0符号なし数値 a >= b
JB / JNAEより小さいときにジャンプ（符号なし）CF = 1符号なし数値 a < b
JBE / JNA以下であるときにジャンプ（符号なし）CF=1 或 ZF=1符号なし数値 a <= b

多くの命令には2つの別名があり（JE と JZ など）、それらは機械語レベルでは完全に同一である。どちらを使うかは文脈による：比較の後では JE/JNE、演算結果の検査の後では JZ/JNZ を使う。これによりコードの可読性が向上する。

単分岐 IF 構造

实例

; 文件路径：if_demo.asm
; 实现：if (x > 10) x = 10;

section .data
    x dd 15                     ; 测试值
    limit dd 10

section .text
    global _start

_start:
    mov eax, [x]                ; 加载 x 到 eax
    cmp eax, [limit]            ; x > 10 ?
    jle skip_update             ; 如果 x <= 10，跳过更新

    mov dword [x], 10           ; x = 10

skip_update:
    ; 程序继续...（这里 x 已经是 10 了）

    mov eax, 1
    mov ebx, 0
    int 0x80

IF-ELSE 二重分岐構造

实例

; 文件路径：if_else_demo.asm
; 实现：if (score >= 60) grade = 'P' else grade = 'F'

section .data
    score dd 75                 ; 考试分数
    grade db 0                  ; 成绩等级
    PASS_SCORE equ 60

section .text
    global _start

_start:
    mov eax, [score]            ; 加载分数
    cmp eax, PASS_SCORE         ; score >= 60 ?
    jge pass_label              ; 如果 >= 60，跳到 pass

    ; else 分支：不及格
    mov byte [grade], 'F'       ; grade = 'F'
    jmp end_if                  ; 跳过 if 分支

pass_label:
    ; if 分支：及格
    mov byte [grade], 'P'       ; grade = 'P'

end_if:
    ; grade 变量现在已设置好

    ; 输出成绩等级
    mov eax, 4
    mov ebx, 1
    mov ecx, grade
    mov edx, 1
    int 0x80

    mov eax, 1
    mov ebx, 0
    int 0x80

运行结果：

$ nasm -f elf32 if_else_demo.asm -o if_else_demo.o
$ ld -m elf_i386 if_else_demo.o -o if_else_demo
$ ./if_else_demo
P

IF-ELSE IF-ELSE 多重分岐構造

次に、点数評価システムを例として、多分岐判断のフローを示す。

实例

; 文件路径：multi_branch.asm
; 分数评级：>=90 -> A, >=80 -> B, >=70 -> C, >=60 -> D, <60 -> F

section .data
    score dd 85
    result db 0
    newline db 0xA

section .text
    global _start

_start:
    mov eax, [score]            ; 加载分数

    ; 检查 >= 90
    cmp eax, 90
    jl check_80                 ; 如果 < 90，继续检查
    mov byte [result], 'A'
    jmp print_result

check_80:
    cmp eax, 80
    jl check_70
    mov byte [result], 'B'
    jmp print_result

check_70:
    cmp eax, 70
    jl check_60
    mov byte [result], 'C'
    jmp print_result

check_60:
    cmp eax, 60
    jl fail_label
    mov byte [result], 'D'
    jmp print_result

fail_label:
    mov byte [result], 'F'

print_result:
    ; 输出评级
    mov eax, 4
    mov ebx, 1
    mov ecx, result
    mov edx, 1
    int 0x80

    ; 输出换行
    mov eax, 4
    mov ebx, 1
    mov ecx, newline
    mov edx, 1
    int 0x80

    mov eax, 1
    mov ebx, 0
    int 0x80

TEST - 非破壊的テスト命令

TESTビットごとのANDを実行するが、結果は保存せず、フラグのみ更新する。

TEST特定のビットが1であるかどうか、またはレジスタが0であるかどうかを調べるためによく使用される。

实例

; TEST 指令示例

    ; 检查 eax 是否为 0（比 CMP eax, 0 更高效）
    test eax, eax               ; eax & eax = eax，只更新 ZF
    jz  eax_is_zero             ; 若 ZF=1，说明 eax=0

    ; 检查特定位
    test al, 0x01               ; 检查 bit0 是否为 1
    jnz bit0_is_set             ; 若 bit0=1，跳转

    ; 检查多个位
    test al, 0x03               ; 检查 bit0 和 bit1 是否有至少一个为 1
    jnz some_bit_set

SETcc - 条件設定命令

ジャンプはせず、条件に応じて対象バイトを1または0に設定する：

实例

; SETcc 条件设置示例

    ; 将 a > b 的结果存入 al
    mov eax, 10
    cmp eax, 5                  ; 10 > 5 ?
    setg al                     ; al = 1（大于成立）
    ; 如果 eax = 3，则 al = 0

    ; 其他 SETcc 指令：
    ; sete / setz   -> 相等/为零
    ; setne / setnz -> 不等/不为零
    ; setl          -> 小于（有符号）
    ; setb          -> 小于（无符号）
    ; setg          -> 大于（有符号）
    ; seta          -> 大于（无符号）

完全な例：偶数・奇数と正・負の判定

实例

; 文件路径：number_check.asm
; 判断数字的奇偶、正负

section .data
    number dd -42
    msg_even db 'Even', 0xA
    msg_even_len equ $ - msg_even
    msg_odd db 'Odd', 0xA
    msg_odd_len equ $ - msg_odd
    msg_pos db 'Positive', 0xA
    msg_pos_len equ $ - msg_pos
    msg_neg db 'Negative', 0xA
    msg_neg_len equ $ - msg_neg
    msg_zero db 'Zero', 0xA
    msg_zero_len equ $ - msg_zero

section .text
    global _start

_start:
    mov eax, [number]

    ; 判断是否为 0
    cmp eax, 0
    jne check_sign
    ; 为零
    mov eax, 4
    mov ebx, 1
    mov ecx, msg_zero
    mov edx, msg_zero_len
    int 0x80
    jmp exit

check_sign:
    ; 判断正负
    cmp eax, 0
    jg  is_positive             ; eax > 0

    ; 负数
    mov eax, 4
    mov ebx, 1
    mov ecx, msg_neg
    mov edx, msg_neg_len
    int 0x80
    jmp check_parity

is_positive:
    ; 正数
    mov eax, 4
    mov ebx, 1
    mov ecx, msg_pos
    mov edx, msg_pos_len
    int 0x80

check_parity:
    ; 判断奇偶（只需看最低位）
    mov eax, [number]
    test eax, 1                 ; 检查 bit0
    jnz is_odd

    ; 偶数
    mov eax, 4
    mov ebx, 1
    mov ecx, msg_even
    mov edx, msg_even_len
    int 0x80
    jmp exit

is_odd:
    ; 奇数
    mov eax, 4
    mov ebx, 1
    mov ecx, msg_odd
    mov edx, msg_odd_len
    int 0x80

exit:
    mov eax, 1
    mov ebx, 0
    int 0x80

条件分岐命令は現在のコードセグメント内のラベルにのみジャンプできる。ジャンプ範囲は通常、約128バイト（短距離ジャンプ）から2GB（近距離ジャンプ）に制限される。NASMは自動的に適切なジャンプエンコーディングを選択する。
