アセンブリ言語 - システムコール

システムコール（System Call）は、ユーザープログラムがオペレーティングシステムのカーネルと対話する唯一の経路です。システムコールを通じて、アセンブリプログラムはファイルの読み書き、メモリの割り当て、プロセスの作成などを行うことができます。

システムコールとは

ユーザープログラムは制限された環境で実行される用户态（User Mode）、ハードウェアに直接アクセスしたり、特権操作を実行したりすることはできません。

プログラムが IO 操作やメモリ割り当てなどのカーネルレベルの機能を実行する必要がある場合、必ず経由しなければなりません。系统调用オペレーティングシステムのカーネルにサービスを要求する。

システムコールの流れは次のとおりです。ユーザープログラムがシステムコールを発行 → CPUがカーネルモードに切り替わる → カーネルが対応するサービスを実行 → ユーザーモードに戻って実行を継続する。

システムコールは次のように理解できます。ユーザープログラムはオペレーティングシステムに電話をかけ、自分には権限のないことを手伝ってもらうよう要求します。

Linuxシステムコールの仕組み

Linux 32ビットシステムでは、システムコールは次の手順で実行されます：

  1. 将 系统调用号EAXレジスタに格納する
  2. パラメータをEBX、ECX、EDX、ESI、EDIなどのレジスタに格納する
  3. 执行int 0x80命令がソフトウェア割り込みを発生させる
  4. CPUがカーネルモードに切り替わり、カーネルはEAX内の呼び出し番号に基づいて対応するサービスを実行する
  5. カーネルは戻り値をEAXに格納し、ユーザーモードに戻って実行を継続する

よく使われるシステムコール一覧

系统调用システムコール番号（EAX）功能参数
sys_exit1退出程序EBX = 返回值（exit code）
sys_fork2创建子进程无参数
sys_read3读取文件/输入EBX=fd, ECX=buf, EDX=count
sys_write4写入文件/输出EBX=fd, ECX=buf, EDX=count
sys_open5打开文件EBX=filename, ECX=flags, EDX=mode
sys_close6关闭文件EBX=fd
sys_creat8创建文件EBX=filename, ECX=mode
sys_lseek19ファイルポインタを移動するEBX=fd, ECX=offset, EDX=whence
sys_brk45データセグメントのサイズを調整する（メモリ割り当て）EBX=新しいアドレス

完全なシステムコールのリストは以下で確認できます/usr/include/asm/unistd_32.h中で確認します。32ビットと64ビットではシステムコール番号が異なることに注意してください。

sys_write - 文字列を出力する

最もよく使われるシステムコールで、画面に内容を出力するために使用されます：

实例

; 文件路径：write_syscall.asm
; 使用 sys_write 输出字符串

section .data
    msg db 'Hello, runoob! Welcome to assembly.', 0xA
    len equ $ - msg

section .text
    global _start

_start:
    ; 调用 sys_write (4)
    mov eax, 4                  ; 系统调用号 4 = sys_write
    mov ebx, 1                  ; 文件描述符 1 = stdout（标准输出）
    mov ecx, msg                ; 出力するデータのアドレス
    mov edx, len                ; 数据长度（字节数）
    int 0x80                    ; 触发系统调用

    ; 成功時、EAX は実際に書き込んだバイト数を返す

    ; 退出程序
    mov eax, 1
    mov ebx, 0
    int 0x80

运行结果：

$ nasm -f elf32 write_syscall.asm -o write_syscall.o
$ ld -m elf_i386 write_syscall.o -o write_syscall
$ ./write_syscall
Hello, runoob! Welcome to assembly.

sys_read - ユーザー入力を読み取る

標準入力（キーボード）からデータを読み取る：

实例

; 文件路径：read_syscall.asm
; 使用 sys_read 读取用户输入并回显

section .bss
    buffer resb 64              ; 预留 64 字节输入缓冲区

section .data
    prompt db 'Please enter your name: '
    prompt_len equ $ - prompt
    output_msg db 'Hello, '
    output_msg_len equ $ - output_msg

section .text
    global _start

_start:
    ; 输出提示信息
    mov eax, 4
    mov ebx, 1
    mov ecx, prompt
    mov edx, prompt_len
    int 0x80

    ; 读取用户输入
    mov eax, 3                  ; 系统调用号 3 = sys_read
    mov ebx, 0                  ; 文件描述符 0 = stdin（标准输入）
    mov ecx, buffer             ; 输入缓冲区地址
    mov edx, 64                 ; 最多读取 64 字节
    int 0x80

    ; EAX は実際に読み取ったバイト数を返す（末尾の改行を含む）
    mov esi, eax                ; 保存实际输入长度到 esi

    ; 输出 "Hello, "
    mov eax, 4
    mov ebx, 1
    mov ecx, output_msg
    mov edx, output_msg_len
    int 0x80

    ; ユーザー入力の内容を出力する
    mov eax, 4
    mov ebx, 1
    mov ecx, buffer
    mov edx, esi                ; 実際に入力されたバイト数を使用
    int 0x80

    ; 退出程序
    mov eax, 1
    mov ebx, 0
    int 0x80

运行结果：

$ nasm -f elf32 read_syscall.asm -o read_syscall.o
$ ld -m elf_i386 read_syscall.o -o read_syscall
$ ./read_syscall
Please enter your name: runoob
Hello, runoob

sys_exit - プログラムを終了する

すべてのプログラムは正常に終了するために sys_exit を呼び出す必要があります。そうしないと、CPU が後続のゴミデータを実行し続け、セグメンテーションフォルトを引き起こします。

实例

; 使用不同退出码退出

    mov eax, 1                  ; 系统调用号 1 = sys_exit
    mov ebx, 0                  ; 終了コード 0 = 正常終了（0以外の値でも可）
    int 0x80

終了コードは shell 内で経由して確認できます。$?查看：

$ ./program
$ echo $?
0

システムコールの汎用テンプレート

システムコールを作成する際は、以下の一般的なテンプレートに従うことができます：

实例

; 系统调用通用模板
; 适用于 Linux 32 位系统

; 步骤1：将系统调用号放入 EAX
mov eax,系统调用号

; 步骤2：按顺序放入参数
mov ebx, 第1个参数
mov ecx, 第2个参数
mov edx, 第3个参数
mov esi, 第4个参数
mov edi, 第5个参数

; 步骤3：触发系统调用
int 0x80

; ステップ4：戻り値を確認する（EAX 内）
; 通常负值表示错误
cmp eax, 0
jl error_handler              ; 如果 EAX < 0，跳转到错误处理

int 0x80実行後、EAX に戻り値が格納されます。ほとんどのシステムコールは、成功時には非負の値を返し、失敗時には負のエラーコードを返します（たとえば -1 は EPERM を表します）。
