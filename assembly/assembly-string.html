アセンブリ言語 - 文字列処理

文字列処理はプログラミングにおける一般的な要件です。

アセンブリで文字列を処理するということは、メモリ内のバイト列を直接操作することを意味し、より低レベルではありますが、より柔軟でもあります。

文字列の定義と保存

NASM では、文字列は本質的にバイト列であり、DB 疑似命令で定義できます：

实例

; 字符串定义方式

section .data
    ; 方式1：标准字符串
    str1 db 'Hello, サンプル!', 0     ; C 风格：以 null 结尾

    ; 方式2：带换行符
    str2 db 'Line1', 0xA, 'Line2', 0xA

    ; 方式3：反引号支持转义
    str3 db `hello\nworld\n`, 0     ; 自动转换转义序列

    ; 方式4：逐个字符定义
    str4 db 'A', 'B', 'C', 'D', 0

    ; 方式5：使用 dup 生成重复字符
    border db 40 dup('-')           ; 40 个减号

    ; 字符串长度计算（编译时）
    str1_len equ $ - str1           ; 包含结尾 null

文字列の長さを計算する

文字列の長さを取得する方法は2つあります：コンパイル時計算（EQU）と実行時計算（nullを探す走査）です：

实例

; 文件路径：strlen_demo.asm
; 文字列の長さを計算する2つの方法

section .data
    ; 方式A：编译时计算（适用于常量字符串）
    msg1 db 'Hello, サンプル!', 0xA
    msg1_len equ $ - msg1          ; 编译时自动计算

    ; 方式B：null終端の文字列（実行時計算）
    msg2 db 'Find my length', 0    ; 以 0 结尾

section .text
    global _start

_start:
    ; 方式A：コンパイル時に計算された長さを直接使用する
    mov eax, 4
    mov ebx, 1
    mov ecx, msg1
    mov edx, msg1_len              ; 直接使用常量
    int 0x80

    ; 方式B：実行時にnull終端文字列の長さを計算する
    mov esi, msg2                  ; esi 指向字符串起始
    mov ecx, 0                     ; 计数器

strlen_loop:
    cmp byte [esi], 0              ; 現在のバイトは null か？
    je  strlen_done                ; はい、終了
    inc ecx                        ; 计数 +1
    inc esi                        ; 指针 +1
    jmp strlen_loop

strlen_done:
    ; ecx には現在文字列長が格納されている（null は含まない）
    mov eax, 4
    mov ebx, 1
    mov ecx, msg2
    mov edx, ecx                   ; 計算された長さを使用
    ; ここに問題があります：ecx は上書きされました。先に保存すべきです
    ; 正确做法：push ecx 再 pop 到 edx

    mov eax, 1
    mov ebx, 0
    int 0x80

字符串复制

ループを使用して1バイトずつコピーするか、x86の文字列操作命令を使用しますMOVSB：

实例

; 文件路径：strcpy_demo.asm
; 字符串复制两种实现方式

section .data
    src db 'サンプル source string', 0
    src_len equ $ - src

section .bss
    dest_manual resb 64             ; 手动复制目标
    dest_fast resb 64               ; 快速复制目标

section .text
    global _start

_start:
    ; 方式A：手动逐字节复制
    mov esi, src                    ; 源地址
    mov edi, dest_manual            ; 目标地址
    mov ecx, src_len                ; 字节数

copy_loop:
    mov al, [esi]                   ; 读取一字节
    mov [edi], al                   ; 写入一字节
    inc esi                         ; 源指针++
    inc edi                         ; 目标指针++
    loop copy_loop

    ; 方式B：使用字符串操作指令（更快）
    cld                             ; 清除方向标志（DF=0，正向复制）
    mov esi, src                    ; 源地址（ESI）
    mov edi, dest_fast              ; 目标地址（EDI）
    mov ecx, src_len                ; 字节数（ECX）
    rep movsb                       ; 重复执行 movsb ecx 次
    ; rep movsb：while(ecx>0) { [edi]=[esi]; esi++; edi++; ecx--; }

    ; 验证：输出两个复制结果
    mov eax, 4
    mov ebx, 1
    mov ecx, dest_manual
    mov edx, src_len
    int 0x80

    mov eax, 4
    mov ebx, 1
    mov ecx, dest_fast
    mov edx, src_len
    int 0x80

    mov eax, 1
    mov ebx, 0
    int 0x80

REP MOVSBこれはx86で最も古典的な文字列コピー方式です。しかし、現代のCPUでは、手動コピーループにループアンローリングを組み合わせた方がREP MOVSBより速い場合があります。現代のCPUは単純な操作に対してより良いパイプライン最適化を行うからです。

字符串比较

使用CMPSB配合REPE（等しくなるまで繰り返す）1バイトずつ比較します：

实例

; 文件路径：strcmp_demo.asm
; 2つの文字列が等しいかどうかを比較する

section .data
    str_a db 'サンプル', 0
    str_b db 'サンプル', 0
    str_c db 'サンプル', 0
    eq_msg db 'Strings are equal', 0xA
    eq_len equ $ - eq_msg
    ne_msg db 'Strings are NOT equal', 0xA
    ne_len equ $ - ne_msg

section .text
    global _start

_start:
    cld                             ; 正向比较
    mov esi, str_a                  ; 第一个字符串
    mov edi, str_b                  ; 第二个字符串
    mov ecx, 7                      ; 最多比较 7 字节（含 null）

    repe cmpsb                      ; 重复比较直到不等或 ecx=0
    ; repe: 如果 ZF=1（相等）且 ecx>0 则继续
    ; cmpsb: [esi] と [edi] を比較し、その後 esi++, edi++

    je  strings_equal               ; 如果 ZF=1，所有字节都相等

    ; 不等
    mov eax, 4
    mov ebx, 1
    mov ecx, ne_msg
    mov edx, ne_len
    int 0x80
    jmp compare_next

strings_equal:
    mov eax, 4
    mov ebx, 1
    mov ecx, eq_msg
    mov edx, eq_len
    int 0x80

compare_next:
    ; str_a と str_c を比較する（大文字小文字が異なる）
    cld
    mov esi, str_a
    mov edi, str_c
    mov ecx, 7
    repe cmpsb
    jne not_equal_2

    mov eax, 4
    mov ebx, 1
    mov ecx, eq_msg
    mov edx, eq_len
    int 0x80
    jmp exit

not_equal_2:
    mov eax, 4
    mov ebx, 1
    mov ecx, ne_msg
    mov edx, ne_len
    int 0x80

exit:
    mov eax, 1
    mov ebx, 0
    int 0x80

文字列操作命令のまとめ

指令功能使用するレジスタ
MOVSBバイトのコピー：[EDI] = [ESI]ESI=ソース、EDI=デスティネーション、ECX=回数、DF=方向
MOVSWワードのコピー（2バイト）同上
MOVSDダブルワードのコピー（4バイト）同上
STOSBバイトの格納：[EDI] = ALEDI=デスティネーション、AL=値、ECX=回数
LODSBバイトの取得：AL = [ESI]ESI=源
CMPSBバイトの比較：[ESI] - [EDI]ESI=ソース、EDI=デスティネーション、ECX=回数
SCASBバイトのスキャン：AL - [EDI]EDI=デスティネーション、AL=検索値、ECX=回数

大文字小文字変換の例

实例

; 文件路径：case_convert.asm
; 文字列内の小文字を大文字に変換する

section .data
    msg db 'Hello, サンプル! Welcome to Assembly.', 0xA
    len equ $ - msg

section .text
    global _start

_start:
    ; 输出原始字符串
    mov eax, 4
    mov ebx, 1
    mov ecx, msg
    mov edx, len
    int 0x80

    ; 转换：遍历字符串，小写 -> 大写
    mov esi, msg                    ; 指向字符串开头
    mov ecx, len                    ; 循环次数

convert_loop:
    mov al, [esi]                   ; 读取字符
    cmp al, 'a'                     ; >= 'a' かどうか
    jb  next_char                   ; 小于 'a'，跳过
    cmp al, 'z'                     ; <= 'z' かどうか
    ja  next_char                   ; 大于 'z'，跳过

    ; 小写转大写：'a'(97) - 'A'(65) = 32
    sub al, 32                      ; 大文字に変換
    mov [esi], al                   ; 写回

next_char:
    inc esi
    loop convert_loop

    ; 変換後の文字列を出力する
    mov eax, 4
    mov ebx, 1
    mov ecx, msg
    mov edx, len
    int 0x80

    mov eax, 1
    mov ebx, 0
    int 0x80

运行结果：

$ nasm -f elf32 case_convert.asm -o case_convert.o
$ ld -m elf_i386 case_convert.o -o case_convert
$ ./case_convert
Hello, サンプル! Welcome to Assembly.
HELLO, サンプル! WELCOME TO ASSEMBLY.

方向フラグ DF は文字列操作の方向を決定します：DF=0 のとき ESI/EDI は増加（正方向）、DF=1 のときは減少（逆方向）です。使用CLDDFをクリア、使用STDDF をセットします。C 関数またはシステムコールを呼び出す前に、DF=0 を確認する必要があります（cdecl の要件）。
