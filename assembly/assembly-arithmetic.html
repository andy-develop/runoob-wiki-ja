アセンブリ言語 - 算術命令

算術命令はCPUが数学演算を実行する基礎です。この章では、x86アーキテクチャにおける加算、減算、乗算、除算、および関連するキャリー演算命令について詳しく説明します。

ADD - 加算命令

ADDソースオペランドとデスティネーションオペランドを加算し、結果をデスティネーションオペランドに格納します。

实例

; 文件路径：add_demo.asm
; ADD 指令示例：计算 100 + 200 + 300

section .data
    a dd 100
    b dd 200
    c dd 300
    sum dd 0

section .text
    global _start

_start:
    ; 方式1：寄存器 += 立即数
    mov eax, [a]                ; eax = 100
    add eax, 200                ; eax = 100 + 200 = 300

    ; 方式2：寄存器 += 寄存器
    mov ebx, [b]                ; ebx = 200
    add eax, ebx                ; eax = 300 + 200 = 500

    ; 方式3：寄存器 += 内存
    add eax, [c]                ; eax = 500 + 300 = 800

    ; 存储结果
    mov [sum], eax              ; sum = 800

    ; 加算がフラグにとえる影響
    mov eax, 0xFFFFFFFF         ; eax = 最大の32ビット符号なし整数
    add eax, 1                  ; eax = 0（溢出回绕）
    ; CF = 1（产生进位）
    ; ZF = 1（結果は 0）
    ; OF = 0（有符号视角无溢出）

    mov eax, 1
    mov ebx, 0
    int 0x80

SUB - 減算命令

SUBデスティネーションオペランドからソースオペランドを減算し、結果をデスティネーションオペランドに格納します。

实例

; 文件路径：sub_demo.asm
; SUB 指令示例

section .data
    x dd 1000
    y dd 300

section .text
    global _start

_start:
    ; 基本减法
    mov eax, [x]                ; eax = 1000
    sub eax, [y]                ; eax = 1000 - 300 = 700

    ; 減算がフラグにとえる影響
    mov eax, 10
    sub eax, 20                 ; eax = -10（即 0xFFFFFFF6）
    ; CF = 1（产生借位：10 < 20）
    ; SF = 1（結果は負）
    ; ZF = 0（结果非零）
    ; OF = 0（无符号溢出）

    ; 减自身：常用于清零
    mov eax, 12345
    sub eax, eax                ; eax = 0
    ; ZF = 1, CF = 0
    ; これはレジスタをゼロクリアする古典のな方法です（mov eax, 0 よりも効率の）

    mov eax, 1
    mov ebx, 0
    int 0x80

sub eax, eaxレジスタをゼロクリアする古典的なテクニックであり、わずか2バイトしか占有しない。一方、mov eax, 05バイトを占有する。極限までサイズを最適化する必要があるシナリオ（シェルコードなど）でよく使われる。

INC / DEC - 自增/自减指令

ADD/SUB よりも簡潔なインクリメント/デクリメント命令：

实例

; INC と DEC の例

section .data
    counter dd 0

section .text
    global _start

_start:
    mov dword [counter], 0      ; counter = 0
    inc dword [counter]         ; counter = 1（内存操作数）
    inc dword [counter]         ; counter = 2

    mov ecx, 10
    dec ecx                     ; ecx = 9
    dec ecx                     ; ecx = 8

    ; INC/DEC は CF フラグに影響しません（これは ADD/SUB との重要な違いです）
    ; 但会影响 ZF、SF、OF、PF

    ; 循环中使用 INC
    mov ecx, 5
    mov eax, 0
loop_inc:
    inc eax                     ; eax 每次加 1
    loop loop_inc               ; 重复 5 次，eax 最终 = 5

    mov ebx, eax                ; 返回值 = 5
    mov eax, 1
    int 0x80

MUL - 符号なし乗算

MUL符号なし乗算を実行する。乗算の規則はやや特殊である：

操作数大小乘数被乗数（暗黙的）结果存放
1 字节任意の8ビットレジスタまたはメモリALAX = AL × オペランド
2 字节任意の16ビットレジスタまたはメモリAXDX:AX = AX × 操作数
4 字节任意の32ビットレジスタまたはメモリEAXEDX:EAX = EAX × 操作数

实例

; 文件路径：mul_demo.asm
; MUL 无符号乘法示例

section .data
    val1 dd 1000
    val2 dd 2000
    result_low dd 0
    result_high dd 0

section .text
    global _start

_start:
    ; 32 位乘法：EDX:EAX = EAX × 操作数
    mov eax, [val1]             ; eax = 1000
    mul dword [val2]            ; edx:eax = 1000 × 2000 = 2,000,000
    ; 结果 = 2,000,000 = 0x001E8480
    ; EAX = 0x001E8480（低 32 位）
    ; EDX = 0x00000000（上位 32 ビット、結果が 32 ビットを超えなかったため）

    ; 大数乘法演示（结果超过 32 位）
    mov eax, 0xFFFFFFFF         ; eax = 4,294,967,295
    mov ebx, 2                  ; ebx = 2
    mul ebx                     ; edx:eax = 0xFFFFFFFF × 2
    ; EAX = 0xFFFFFFFE
    ; EDX = 0x00000001（高 32 位）
    ; CF = 1（结果超出 32 位）

    ; 保存结果
    mov [result_low], eax
    mov [result_high], edx

    mov eax, 1
    mov ebx, 0
    int 0x80

IMUL - 符号付き乗算

IMUL符号付き数の乗算に使用される。三つの形式がある：

实例

; 文件路径：imul_demo.asm
; IMUL 有符号乘法示例

section .data
    a dd -100
    b dd 3

section .text
    global _start

_start:
    ; 单操作数形式：同 MUL
    mov eax, [a]                ; eax = -100
    imul dword [b]              ; edx:eax = -100 × 3 = -300
    ; EAX = -300（0xFFFFFED4）
    ; EDX = 0xFFFFFFFF（符号扩展）

    ; 双操作数形式：reg = reg × 操作数
    mov ebx, [a]                ; ebx = -100
    imul ebx, [b]               ; ebx = -100 × 3 = -300
    ; 結果は32ビットに収まる必要がある

    ; 三操作数形式：reg = 操作数1 × 立即数
    imul ecx, [a], 5            ; ecx = -100 × 5 = -500

    mov eax, 1
    mov ebx, 0
    int 0x80

MULとIMUL主な違いは符号の扱いです：MUL符号なし数として解釈され、IMUL符号付き数値として解釈される。例えば、0xFFFFFFFF は MUL では 4294967295 であり、IMUL では -1 である。

DIV - 符号なし除算

DIV符号なし除算を実行する。規則は MUL と対称的である：

除数大小被除数（暗黙的）商余数
1 字节AXALAH
2 字节DX:AXAXDX
4 字节EDX:EAXEAXEDX

实例

; 文件路径：div_demo.asm
; DIV 无符号除法示例

section .data
    dividend dd 1000
    divisor dd 7
    quotient dd 0
    remainder dd 0

section .text
    global _start

_start:
    ; 32 位除法：edx:eax / 除数
    ; 被除数をあらかじめ EDX:EAX に拡張する必要がある
    mov eax, [dividend]         ; eax = 1000
    mov edx, 0                  ; edx = 0（高 32 位清零）
    div dword [divisor]         ; edx:eax / 7
    ; EAX = 142（商）
    ; EDX = 6（余数：1000 = 142×7 + 6）

    mov [quotient], eax         ; quotient = 142
    mov [remainder], edx        ; remainder = 6

    ; 字节除法示例：55 / 4
    mov ax, 55                  ; 被除数
    mov bl, 4                   ; 除数
    div bl                      ; al = 13（商）, ah = 3（余数）

    mov eax, 1
    mov ebx, 0
    int 0x80

除算を行う前に必ず用いるmov edx, 0 或 xor edx, edxEDX をゼロクリアしよう！ EDX に古いデータが入っていると、被除数の値が正しくなくなる。これは初心者に最もよくある除算バグである。

IDIV - 符号付き除算

IDIV符号付き除算に使用する。除算の前には、使用する必要があるCDQEAX を EDX:EAX に符号拡張する命令：

实例

; IDIV 有符号除法示例

    ; 计算 -100 / 3
    mov eax, -100               ; eax = -100
    cdq                         ; 符号扩展：edx:eax = -100
                                ; cdq は eax の符号ビットを edx の全ビットにコピーする
    mov ebx, 3                  ; 除数
    idiv ebx                    ; eax = -33（商）, edx = -1（余数）
    ; 验证：-100 = -33 × 3 + (-1)

ADC / SBB - 带进位/借位运算

用于大数运算（32ビットを超えるデータ）：

实例

; 文件路径：adc_sbb_demo.asm
; 64 位加法：两个 64 位数相加

section .data
    ; 64 位数 a = 0x00000001 FFFFFFFF
    a_low dd 0xFFFFFFFF
    a_high dd 0x00000001

    ; 64 位数 b = 0x00000000 00000005
    b_low dd 0x00000005
    b_high dd 0x00000000

    ; 结果
    result_low dd 0
    result_high dd 0

section .text
    global _start

_start:
    ; 低 32 位相加
    mov eax, [a_low]
    add eax, [b_low]            ; eax = 0xFFFFFFFF + 5 = 0x00000004
    ; CF = 1（产生进位！）
    mov [result_low], eax       ; result_low = 0x00000004

    ; 高 32 位加进位
    mov eax, [a_high]
    adc eax, [b_high]           ; adc = add + CF
    ; eax = 1 + 0 + 1(进位) = 2
    mov [result_high], eax      ; result_high = 2

    ; 最终 64 位结果：0x00000002 00000004
    ; 验证：0x1FFFFFFF + 5 = 0x200000004

    ; SBB 减法同理（带借位）
    mov eax, [a_low]
    sub eax, [b_low]            ; 低 32 位相减
    mov [result_low], eax

    mov eax, [a_high]
    sbb eax, [b_high]           ; sbb = sub - CF

    mov eax, 1
    mov ebx, 0
    int 0x80
