アセンブリ言語 - ループ構造

ループはプログラムで最も一般的な制御構造の一つです。アセンブリ言語では、LOOP命令や条件分岐を使って様々なループロジックを実装できます。

LOOP 指令

LOOPx86がループ専用に設計した命令で、ECXをカウンタとして使用します：

LOOPを実行するたびに、ECXは自動的に1減算され、ECXが0でなければターゲットラベルにジャンプします。

实例

; 文件路径：loop_basic.asm
; LOOP 指令基本用法：重复 5 次输出

section .data
    msg db 'サンプル', 0xA
    len equ $ - msg

section .text
    global _start

_start:
    mov ecx, 5                  ; 循环计数器 = 5（循环 5 次）

repeat:
    ; 保存 ecx（系统调用可能修改它）
    push ecx

    ; 输出 msg
    mov eax, 4
    mov ebx, 1
    mov ecx, msg
    mov edx, len
    int 0x80

    ; 恢复 ecx
    pop ecx
    loop repeat                 ; ecx--; if ecx != 0 跳转到 repeat

    mov eax, 1
    mov ebx, 0
    int 0x80

运行结果：

$ nasm -f elf32 loop_basic.asm -o loop_basic.o
$ ld -m elf_i386 loop_basic.o -o loop_basic
$ ./loop_basic

loopデフォルトではECX（32ビット）をカウンタとして使用します。16ビットモードではCX、64ビットモードではRCXを使用します。loopの前にECXの値を正しく設定してください。

LOOP の変種

指令跳转条件说明
LOOPECX != 0標準的なループでは、先にECXを減算してから判定します。
LOOPE / LOOPZECX != 0 且 ZF = 1等しい場合にループを継続
LOOPNE / LOOPNZECX != 0 且 ZF = 0等しくない場合にループを継続

实例

; LOOPE の例：配列内で最初の非ゼロ値を検索する

section .data
    array db 0, 0, 0, 5, 0, 0  ; 最初の3つは 0、4番目は 5

section .text
    global _start

_start:
    mov ecx, 6                  ; 数组长度
    mov esi, array - 1          ; 指向数组前一个位置

find_nonzero:
    inc esi                     ; 移动到下一个元素
    cmp byte [esi], 0           ; 当前元素 == 0 ?
    loope find_nonzero          ; 0 かつ ecx > 0 の場合は、ループを続ける
    ; ループ終了：最初の非ゼロ要素が見つかった、または走査が終了した

    ; esi は現在、最初の非ゼロ要素のアドレスを指している
    ; ecx 未比較の残りの要素数

    mov eax, 1
    mov ebx, 0
    int 0x80

条件ループ（WHILE ループ）

条件分岐でwhileループを実装する：先に条件を判定し、それからループ本体を実行します。

实例

; 文件路径：while_loop.asm
; 实现：while (x < 100) x = x * 2;

section .data
    x dd 1
    limit dd 100

section .text
    global _start

_start:
    ; while 循环
while_start:
    mov eax, [x]                ; 加载 x
    cmp eax, [limit]            ; x < 100 ?
    jge while_end               ; 如果 x >= 100，结束循环

    shl eax, 1                  ; x = x * 2
    mov [x], eax                ; 存回 x

    jmp while_start             ; 回到循环开始，重新判断

while_end:
    ; x は現在 128 です (1->2->4->8->16->32->64->128、128 >= 100 で停止)

    mov eax, 1
    mov ebx, [x]                ; x を終了コードとして返す
    int 0x80

DO-WHILE 循环

まずループ本体を実行し、それから条件を判断する：

实例

; DO-WHILE 循环：至少执行一次

section .data
    counter dd 0

section .text
    global _start

_start:
    mov dword [counter], 0

do_loop:
    ; 循环体：counter++（至少执行一次）
    inc dword [counter]

    ; 判断条件
    cmp dword [counter], 5
    jl  do_loop                 ; counter < 5 时继续
    ; counter 最终 = 5

    mov eax, 1
    mov ebx, 0
    int 0x80

嵌套循环

ネストされたループでは、外側のカウンタの保存に注意する必要があります：

实例

; 文件路径：nested_loop.asm
; 嵌套循环示例：打印乘法表（3×3）

section .data
    space db ' '
    newline db 0xA
    ; 変換後の数字文字の格納用（2桁 + スペース + null）
    num_str db '  ', 0

section .text
    global _start

_start:
    mov ecx, 3                  ; 外层循环计数器（行数）

outer_loop:
    push ecx                    ; ★ 保存外层计数器！（重要）
    mov ecx, 3                  ; 内层循环计数器（列数）

inner_loop:
    push ecx                    ; 保存内层计数器

    ; 计算乘积 = 外层行号(4-当前ecx) × 内层列号
    ; 此处省略具体数字转换输出，仅演示结构
    ; 实际输出：行号×列号

    pop ecx                     ; 恢复内层计数器
    loop inner_loop             ; 内层循环

    ; 输出换行
    mov eax, 4
    mov ebx, 1
    mov ecx, newline
    mov edx, 1
    int 0x80

    pop ecx                     ; ★ 恢复外层计数器
    loop outer_loop             ; 外层循环

    mov eax, 1
    mov ebx, 0
    int 0x80

ネストされたループで最も一般的なエラーは、スタックを使って外側のカウンタの値を保存するのを忘れることです。LOOPはECXを変更するため、内側のループでECXを再設定すると、外側のカウンタ値が失われます。解決方法は、内側のループに入るたびに（前に値を保存することです）。push ecx，离开后pop ecx。

ループ最適化テクニック

ループの効率を向上させるいくつかの方法：

实例

; ループ最適化の比較

; 方式A：使用 LOOP 指令（较慢，但不明显）
    mov ecx, 1000
loop_a:
    add eax, [ebx]
    add ebx, 4
    loop loop_a

; 方式B：使用条件跳转（手工计数）（通常更快）
    mov ecx, 1000
loop_b:
    add eax, [ebx]
    add ebx, 4
    dec ecx
    jnz loop_b

; 方式C：循环展开（最快，但代码体积大）
    ; 1000 回のループを、1 回あたり 4 要素ずつ処理するように展開する
    mov ecx, 250                ; 1000 / 4 = 250
loop_unrolled:
    add eax, [ebx]
    add eax, [ebx + 4]
    add eax, [ebx + 8]
    add eax, [ebx + 12]
    add ebx, 16
    dec ecx
    jnz loop_unrolled

; 方式D：倒计数（减少比较指令）
    mov ecx, 1000
    lea ebx, [array + 1000*4 - 4]  ; 配列の末尾から開始
loop_reverse:
    add eax, [ebx]
    sub ebx, 4
    dec ecx
    jnz loop_reverse

完全な例：1から100までの合計を計算します。

实例

; 文件路径：sum_1_to_100.asm
; 计算 1+2+...+100 = 5050

section .data
    msg db 'Sum of 1 to 100 is: '
    msg_len equ $ - msg
    newline db 0xA

section .bss
    result_str resb 12           ; 変換後の数値文字列を格納する

section .text
    global _start

_start:
    ; 累積とを計算
    mov ecx, 100                 ; 循环 100 次
    mov eax, 0                   ; 累積との初期値は 0
    mov ebx, 1                   ; 現在加算する数

sum_loop:
    add eax, ebx                 ; eax += ebx
    inc ebx                      ; 下一个数
    loop sum_loop                ; 继续循环
    ; eax = 5050

    ; 输出提示文字
    push eax                     ; 保存结果
    mov eax, 4
    mov ebx, 1
    mov ecx, msg
    mov edx, msg_len
    int 0x80
    pop eax                      ; 恢复结果

    ; 数値を文字列に変換する（簡易版：整数値を直接出力する）
    ; ここでは簡略化して、eax の下位バイトを文字に変換する
    add al, '0'                  ; 不准确，仅演示
    mov [result_str], al

    ; 输出结果
    mov eax, 4
    mov ebx, 1
    mov ecx, result_str
    mov edx, 12
    int 0x80

    mov eax, 4
    mov ebx, 1
    mov ecx, newline
    mov edx, 1
    int 0x80

    mov eax, 1
    mov ebx, 0
    int 0x80

数値を文字列に変換するのはアセンブリでよくある問題です。上記の簡単な方法は、数値が0-9の場合にのみ有効です。完全な数値から文字列への変換アルゴリズムは「数値処理」の章で詳しく説明します。
