アセンブリ言語 - 定数

定数はコンパイル時に決定され、プログラム実行中に変更できない値です。アセンブリで定数を使用すると、コードがより読みやすく、保守しやすくなります。

アセンブリにおける定数とは

NASM では、定数は伪指令定義されたシンボルは、アセンブラがコンパイル時にすべての定数参照を実際の値に置き換えます。

変数とは異なり、定数はデータセグメントのメモリを占有しません。それらは编译时置き換えられ、C言語の...に相当します。#define。

EQU - 等価定数

EQUは最も一般的な定数定義方法であり、定義後は再定義できません：

实例

; EQU 常量定义示例

; 数值常量
MAX_SIZE equ 256                 ; 缓冲区最大大小
DEFAULT_PORT equ 8080            ; 默认端口号
TRUE equ 1                       ; 真值
FALSE equ 0                      ; 假值

; 字符常量
LF equ 0xA                       ; 换行符
CR equ 0xD                       ; 回车符
NULL equ 0                       ; 空字符

; 表达式常量
ARRAY_SIZE equ 100 * 4           ; 100個のダブルワードのバイト数
BUFFER_SIZE equ MAX_SIZE * 2     ; 引用另一个常量
TOTAL equ 10 + 20 + 30           ; 计算结果

section .data
    ; 使用常量
    buffer db MAX_SIZE dup(0)    ; MAX_SIZE バイトのバッファを定義する

section .text
    global _start

_start:
    mov eax, MAX_SIZE            ; eax = 256
    mov ebx, BUFFER_SIZE         ; ebx = 512
    mov ecx, ARRAY_SIZE          ; ecx = 400

    mov eax, 1
    mov ebx, 0
    int 0x80

EQU定義された定数はプログラム内で変更できず、再定義もできません。再定義可能な定数（マクロ変数のような）が必要な場合は、...を使用します。%define。

%define - マクロ定義定数

%defineEQU よりも柔軟で、再定義とスコープ制御をサポートします：

实例

; %define 定数定義と再定義の例

%define VERSION 1                ; 定义版本号
%define APP_NAME 'runoob'        ; 定义应用名称

; %define を使用して定義した定数
    mov eax, VERSION             ; eax = 1

; 再定義が可能（EQU では許可されていません）
%define VERSION 2                ; 版本升级
    mov eax, VERSION             ; 現在 eax = 2

; %undef 取消定义
%undef VERSION                   ; VERSION 不再有效
; mov eax, VERSION               ; 错误：VERSION 未定义

; %define は複数行（括弧付きマクロ）を定義できる
%define sum(a, b) (a + b)
    mov eax, sum(10, 20)         ; mov eax, (10+20) -> eax = 30
特性EQU%define
再定義は可能か否である
可取消定义否である（%undef）
支持参数否はい（マクロ関数）
使用场景真の定数（不変値）可変の設定、マクロ置換

%assign - 代入可能な定数

%assignコンパイル時に複数回変更できる数値定数を定義するために使用されます：

实例

; %assign 编译时可变常量

%assign counter 0                ; 初期値は 0

section .data
    ; 配合 %rep 重复块使用
    ; 生成 table_0, table_1, ..., table_9
    %rep 10
        db counter               ; 使用当前 counter 值
        %assign counter counter + 1
    %endrep

プログラムにおける定数の実際の応用

实例

; 文件路径：constants_practice.asm
; 综合使用常量提高代码可读性

; 系统调用号常量
SYS_EXIT equ 1
SYS_FORK equ 2
SYS_READ equ 3
SYS_WRITE equ 4
SYS_OPEN equ 5
SYS_CLOSE equ 6

; 文件描述符常量
STDIN equ 0
STDOUT equ 1
STDERR equ 2

; 程序常量
MAX_INPUT equ 256
EXIT_SUCCESS equ 0
EXIT_FAILURE equ 1

section .data
    msg db 'Hello, RUNOOB!', 0xA
    msg_len equ $ - msg

section .bss
    buffer resb MAX_INPUT        ; 使用常量定义缓冲区大小

section .text
    global _start

_start:
    ; 数値の代わりに定数名を使用することで、コードの意図が一目瞭然です
    mov eax, SYS_WRITE           ; 明確：これは書き込み操作
    mov ebx, STDOUT              ; 清晰：输出到屏幕
    mov ecx, msg
    mov edx, msg_len
    int 0x80

    mov eax, SYS_EXIT            ; 清晰：退出程序
    mov ebx, EXIT_SUCCESS        ; 清晰：正常退出
    int 0x80

定数名を「マジックナンバー」の代わりに使用することは良い習慣です。3か月後、mov eax, SYS_WRITE 比 mov eax, 4より理解しやすい。

字符串常量

NASM は文字列定数を定義するいくつかの方法をサポートしています：

实例

; 文字列定数の定義方法

section .data
    ; 直接定义（可用 EQU 表示长度）
    s1 db 'hello', 0
    s1_len equ $ - s1

    ; 带转义字符
    s2 db 'Line1', 0xA, 'Line2', 0   ; 0xA は改行文字

    ; 使用反引号支持 C 风格转义
    s3 db `hello\nworld\n`, 0         ; \n は 0xA に自動的に置き換えられる

    ; 多行字符串
    s4 db 'first line', 0xA
       db 'second line', 0xA
       db 'third line', 0

NASM のバッククォート文字列サポート\n（换行）、\t（タブ文字）、\\（バックスラッシュ）、\'（シングルクォート）などの標準Cエスケープシーケンス。
