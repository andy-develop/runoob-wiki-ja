二分探索木の深さ優先走査

二分探索木の走査は、深さ優先走査とレベル順走査の2つの大きな種類に分けられます。

深さ優先走査は、前順走査（preorder tree walk）、中順走査（inorder tree walk）、後順走査（postorder tree walk）の3つに分けられます。それぞれ次のとおりです。

  • 1、前順走査：まず現在のノードを訪問し、次に左右の部分木を順に再帰的に訪問します。
  • 2、中順走査：まず左部分木を再帰的に訪問し、次に自身を訪問し、最後に右部分木を再帰的に訪問します。
  • 3、後順走査：まず左右の部分木を再帰的に訪問し、次に自身のノードを訪問します。

前順走査の結果図：

対応するコード例：

...
// nodeを根とする二分探索木に対して前順走査を行う、再帰アルゴリズム
private void preOrder(Node node){

    if( node != null ){
        System.out.println(node.key);
        preOrder(node.left);
        preOrder(node.right);
    }
}

src/サンプル/binary/Traverse.java のコード：

package サンプル.binary;

/**
* 優先走査
 */

public class Traverse<Key extends Comparable<Key>, Value>  {

    // ツリーのノードはプライベートなクラスで、外部は二分探索木のノードの具体的な実装を知る必要はありません。
    private class Node {
        private Key key;
        private Value value;
        private Node left, right;

        public Node(Key key, Value value) {
            this.key = key;
            this.value = value;
            left = right = null;
        }
    }

    private Node root;  // ルートノード
    private int count;  // ツリー内のノード数

    // コンストラクタ、デフォルトでは空の二分探索木を構築する
    public Traverse() {
        root = null;
        count = 0;
    }

    // 二分探索木のノード数を返す
    public int size() {
        return count;
    }

    // 二分探索木が空かどうかを返す
    public boolean isEmpty() {
        return count == 0;
    }

    // 二分探索木に新しい(key, value)データペアを挿入する
    public void insert(Key key, Value value){
        root = insert(root, key, value);
    }

    // 二分探索木にキーkeyが存在するかどうかを確認する
    public boolean contain(Key key){
        return contain(root, key);
    }

    // 二分探索木内でキーkeyに対応する値を検索する。この値が存在しない場合はnullを返す
    public Value search(Key key){
        return search( root , key );
    }

    // 二分探索木の前順走査
    public void preOrder(){
        preOrder(root);
    }

    // 二分探索木の中順走査
    public void inOrder(){
        inOrder(root);
    }

    // 二分探索木の後順走査
    public void postOrder(){
        postOrder(root);
    }

    //********************
    //* 二分探索木の補助関数
    //********************

    // 再帰アルゴリズムを使用して、nodeを根とする二分探索木にノード(key, value)を挿入する
    // 新しいノードを挿入した後の二分探索木の根を返す
    private Node insert(Node node, Key key, Value value){

        if( node == null ){
            count ++;
            return new Node(key, value);
        }

        if( key.compareTo(node.key) == 0 )
            node.value = value;
        else if( key.compareTo(node.key) < 0 )
            node.left = insert( node.left , key, value);
        else    // key > node->key
            node.right = insert( node.right, key, value);

        return node;
    }

    // 再帰アルゴリズムを使用して、nodeを根とする二分探索木にキー値keyのノードが含まれているかどうかを確認する
    private boolean contain(Node node, Key key){

        if( node == null )
            return false;

        if( key.compareTo(node.key) == 0 )
            return true;
        else if( key.compareTo(node.key) < 0 )
            return contain( node.left , key );
        else // key > node->key
            return contain( node.right , key );
    }

    // 再帰アルゴリズムを使用して、nodeを根とする二分探索木内でkeyに対応するvalueを検索する
    // valueが存在しない場合はNULLを返す
    private Value search(Node node, Key key){

        if( node == null )
            return null;

        if( key.compareTo(node.key) == 0 )
            return node.value;
        else if( key.compareTo(node.key) < 0 )
            return search( node.left , key );
        else // key > node->key
            return search( node.right, key );
    }

    // nodeを根とする二分探索木に対して前順走査を行う、再帰アルゴリズム
    private void preOrder(Node node){

        if( node != null ){
            System.out.println(node.key);
            preOrder(node.left);
            preOrder(node.right);
        }
    }

    // nodeを根とする二分探索木に対して中順走査を行う、再帰アルゴリズム
    private void inOrder(Node node){

        if( node != null ){
            inOrder(node.left);
            System.out.println(node.key);
            inOrder(node.right);
        }
    }

    // nodeを根とする二分探索木に対して後順走査を行う、再帰アルゴリズム
    private void postOrder(Node node){

        if( node != null ){
            postOrder(node.left);
            postOrder(node.right);
            System.out.println(node.key);
        }
    }
}
